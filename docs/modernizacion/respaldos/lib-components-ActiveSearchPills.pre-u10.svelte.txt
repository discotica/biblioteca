<script lang="ts">
  import { afterUpdate, createEventDispatcher, onMount } from 'svelte';

  export type SearchPill = {
    id: string;
    label: string;
    removeLabel: string;
    kind: 'country' | 'genre' | 'text' | 'year' | 'type' | 'radar';
    value?: unknown;
  };

  export let pills: SearchPill[] = [];

  const dispatch = createEventDispatcher<{
    remove: SearchPill;
    overflow: HTMLButtonElement;
  }>();

  let row: HTMLDivElement;
  let measureRack: HTMLDivElement;
  let overflowTrigger: HTMLButtonElement;
  let visibleCount = pills.length;
  let frame = 0;

  $: hiddenCount = Math.max(0, pills.length - visibleCount);
  $: signature = pills.map((pill) => `${pill.id}:${pill.label}`).join('|');
  $: if (signature) scheduleMeasure();
  $: if (!pills.length) visibleCount = 0;

  function scheduleMeasure() {
    if (typeof window === 'undefined') return;
    cancelAnimationFrame(frame);
    frame = requestAnimationFrame(measure);
  }

  function measure() {
    if (!row || !measureRack) return;
    const available = row.clientWidth;
    const widths = Array.from(measureRack.querySelectorAll<HTMLElement>('[data-pill-measure]'))
      .map((node) => node.getBoundingClientRect().width);
    const gap = 6;
    const totalWidth = widths.reduce((sum, width) => sum + width, 0) + Math.max(0, widths.length - 1) * gap;

    if (totalWidth <= available) {
      if (visibleCount !== pills.length) visibleCount = pills.length;
      return;
    }

    // The overflow control has a stable 44 px target. Reserve it before
    // choosing the largest prefix that can remain in the single visible row.
    const overflowWidth = 44;
    let used = overflowWidth;
    let nextVisible = 0;
    for (const width of widths) {
      const next = used + gap + width;
      if (next > available) break;
      used = next;
      nextVisible += 1;
    }
    if (visibleCount !== nextVisible) visibleCount = nextVisible;
  }

  onMount(() => {
    const observer = new ResizeObserver(scheduleMeasure);
    observer.observe(row);
    observer.observe(measureRack);
    scheduleMeasure();
    return () => {
      observer.disconnect();
      cancelAnimationFrame(frame);
    };
  });

  afterUpdate(scheduleMeasure);
</script>

{#if pills.length}
  <div class="pills-shell">
    <div class="pills-row" bind:this={row} aria-label="Filtros activos">
      {#each pills.slice(0, visibleCount) as pill (pill.id)}
        <span class="search-pill chip-activo">
          <span class="search-pill__label">{pill.label}</span>
          <button
            type="button"
            aria-label={pill.removeLabel}
            on:click={() => dispatch('remove', pill)}
          >×</button>
        </span>
      {/each}
      {#if hiddenCount > 0}
        <button
          class="pills-overflow"
          type="button"
          bind:this={overflowTrigger}
          aria-label="Mostrar {hiddenCount} filtros más"
          on:click={() => dispatch('overflow', overflowTrigger)}
        >+{hiddenCount}</button>
      {/if}
    </div>

    <div class="pills-measure" bind:this={measureRack} aria-hidden="true">
      {#each pills as pill (pill.id)}
        <span class="search-pill search-pill--measure" data-pill-measure>
          <span class="search-pill__label search-pill__label--measure" data-label={pill.label}></span><span class="search-pill__measure-close">×</span>
        </span>
      {/each}
    </div>
  </div>
{/if}
