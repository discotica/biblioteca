<script>
  // U10: sin caja propia (la pone el panel o la hoja) y todos los controles
  // con la altura y el radio del sistema. U10b: solo los controles; el campo
  // de texto es SearchCombobox. Una grilla sin huecos: en el panel,
  // País · Género / Tipo · Año · Reiniciar; en la hoja, País · Género /
  // Tipo · Año y Reiniciar en su propia línea.
  import Icon from '$lib/ui/Icon.svelte';
  import { createEventDispatcher, onMount } from 'svelte';
  import { uniqueCountries, uniqueGenres, uniqueTypes, uniqueYears } from '$lib/domain/filterOptions';
  import ResetFiltersButton from '$lib/components/home/ResetFiltersButton.svelte';

  export let valores = {
    pais: [],
    genero: [],
    texto: [],
    año: '',
    type: ''
  };

  export let lanzamientos = [];
  /** 'panel' (desktop, 4 columnas) | 'sheet' (hoja móvil, 2 columnas). */
  export let layout = 'panel';
  /** La hoja lleva «Reiniciar» en su pie hasta U10b-2. */
  export let showReset = true;
  const dispatch = createEventDispatcher();

  $: paisesUnicos = uniqueCountries(lanzamientos);
  $: generosUnicos = uniqueGenres(lanzamientos);
  $: tiposUnicos = uniqueTypes(lanzamientos);
  $: añosUnicos = uniqueYears(lanzamientos);

  function updateCampo(key, value) {
    const nuevo = { ...valores, [key]: value };
    dispatch('filtrar', nuevo);
  }

  function toggleEnLista(key, valor) {
    const yaEstá = valores[key].includes(valor);
    const nuevo = {
      ...valores,
      [key]: yaEstá ? valores[key].filter(v => v !== valor) : [...valores[key], valor]
    };
    dispatch('filtrar', nuevo);
  }

  let detPais, detGenero;

  function handleClickOutside(e) {
    if (detPais && !detPais.contains(e.target)) detPais.removeAttribute('open');
    if (detGenero && !detGenero.contains(e.target)) detGenero.removeAttribute('open');
  }

  onMount(() => {
    document.addEventListener('click', handleClickOutside);
    return () => document.removeEventListener('click', handleClickOutside);
  });
</script>

<section class="filtros-panel">
  <div class="filtros-grilla filtros-grilla--{layout}">
    <!-- País y Género: nubes de etiquetas plegables -->
    <div class="cloud-campo">
      <span class="campo-label">País</span>
      <details class="cloud-dropdown" bind:this={detPais}>
        <summary class="cloud-summary">
          <span class="cloud-text">
            {#if valores.pais.length > 0}{valores.pais.join(', ')}{:else}Seleccionar{/if}
          </span>
          {#if valores.pais.length > 0}
            <span class="cloud-count">{valores.pais.length}</span>
          {/if}
          <span class="campo-chevron"><Icon name="chevron-abajo" size="sm" /></span>
        </summary>
        <div class="cloud-body">
          <div class="tag-cloud">
            {#each paisesUnicos as pais}
              <button
                type="button"
                class="tag"
                title={pais}
                class:tag--active={valores.pais.includes(pais)}
                on:click={() => toggleEnLista('pais', pais)}
              >{pais}</button>
            {/each}
          </div>
        </div>
      </details>
    </div>

    <div class="cloud-campo">
      <span class="campo-label">Género</span>
      <details class="cloud-dropdown" bind:this={detGenero}>
        <summary class="cloud-summary">
          <span class="cloud-text">
            {#if valores.genero.length > 0}{valores.genero.join(', ')}{:else}Seleccionar{/if}
          </span>
          {#if valores.genero.length > 0}
            <span class="cloud-count">{valores.genero.length}</span>
          {/if}
          <span class="campo-chevron"><Icon name="chevron-abajo" size="sm" /></span>
        </summary>
        <div class="cloud-body">
          <div class="tag-cloud">
            {#each generosUnicos as genero}
              <button
                type="button"
                class="tag"
                title={genero}
                class:tag--active={valores.genero.includes(genero)}
                on:click={() => toggleEnLista('genero', genero)}
              >{genero}</button>
            {/each}
          </div>
        </div>
      </details>
    </div>

    <!-- Tipo y Año -->
    <div class="campo">
      <span class="campo-label">Tipo</span>
      <details class="dropdown">
        <summary><span class="campo-valor">{valores.type || 'Seleccionar'}</span><span class="campo-chevron"><Icon name="chevron-abajo" size="sm" /></span></summary>
        <ul>
          <li>
            <button type="button" on:click={() => updateCampo('type', '')}>
              Todos {#if !valores.type}<span class="tick"><Icon name="check" size="em" weight="bold" label="Seleccionado" /></span>{/if}
            </button>
          </li>
          {#each tiposUnicos as tipo}
            <li>
              <button type="button" on:click={() => updateCampo('type', tipo)}>
                {tipo}{#if valores.type === tipo} <span class="tick"><Icon name="check" size="em" weight="bold" label="Seleccionado" /></span>{/if}
              </button>
            </li>
          {/each}
        </ul>
      </details>
    </div>

    <div class="campo">
      <span class="campo-label">Año</span>
      <details class="dropdown">
        <summary><span class="campo-valor">{valores.año || 'Seleccionar'}</span><span class="campo-chevron"><Icon name="chevron-abajo" size="sm" /></span></summary>
        <ul>
          <li>
            <button type="button" on:click={() => updateCampo('año', '')}>
              Todos {#if !valores.año}<span class="tick"><Icon name="check" size="em" weight="bold" label="Seleccionado" /></span>{/if}
            </button>
          </li>
          {#each añosUnicos as año}
            <li>
              <button type="button" on:click={() => updateCampo('año', año)}>
                {año}{#if String(valores.año) === String(año)} <span class="tick"><Icon name="check" size="em" weight="bold" label="Seleccionado" /></span>{/if}
              </button>
            </li>
          {/each}
        </ul>
      </details>
    </div>

    {#if showReset}
      <div class="filtros-reset">
        <ResetFiltersButton block={layout === 'panel'} on:click={() => dispatch('reset')} />
      </div>
    {/if}
  </div>
</section>
