<script>
  // Hoja de filtros móvil (E15). No tiene estado de búsqueda propio: recibe
  // los valores del SearchState y emite los cambios. El gesto de cierre solo
  // empieza en el asa o la cabecera.
  import { createEventDispatcher } from 'svelte';
  import { fade } from 'svelte/transition';
  import Filtros from '$lib/components/Filtros.svelte';
  import { sheetOverlay } from '$lib/ui/overlay';
  import RadarPanel from './RadarPanel.svelte';
  import ResetFiltersButton from './ResetFiltersButton.svelte';

  export let open = false;
  export let pills = [];
  export let lanzamientos = [];
  export let valores;
  export let viewportHeight = 0;
  export let viewportBottom = 0;
  export let returnFocus = null;
  export let radarRef = null;

  const dispatch = createEventDispatcher();

  let swipeOffset = 0;
  let swipeActive = false;
  let swipeCancelled = false;
  let swipePointerId = -1;
  let swipeStartX = 0;
  let swipeStartY = 0;
  let swipeStartAt = 0;

  function close() {
    swipeOffset = 0;
    swipeActive = false;
    dispatch('close');
  }

  function isInteractiveSwipeTarget(target) {
    return target instanceof Element && Boolean(target.closest('button, a, input, select, textarea, summary'));
  }

  function iniciarSwipe(event) {
    if (event.button !== undefined && event.button !== 0) return;
    if (isInteractiveSwipeTarget(event.target)) return;
    swipePointerId = event.pointerId;
    swipeStartX = event.clientX;
    swipeStartY = event.clientY;
    swipeStartAt = performance.now();
    swipeOffset = 0;
    swipeActive = true;
    swipeCancelled = false;
    event.currentTarget.setPointerCapture?.(event.pointerId);
  }

  function moverSwipe(event) {
    if (!swipeActive || event.pointerId !== swipePointerId) return;
    const dx = event.clientX - swipeStartX;
    const dy = event.clientY - swipeStartY;
    if (Math.abs(dx) > 10 && Math.abs(dx) > Math.abs(dy)) {
      swipeCancelled = true;
      swipeOffset = 0;
      return;
    }
    if (swipeCancelled) return;
    swipeOffset = Math.max(0, dy);
  }

  function terminarSwipe(event) {
    if (!swipeActive || event.pointerId !== swipePointerId) return;
    const elapsed = Math.max(1, performance.now() - swipeStartAt);
    const velocity = swipeOffset / elapsed;
    const shouldClose = !swipeCancelled && (swipeOffset >= 72 || (swipeOffset >= 24 && velocity >= 0.5));
    swipeActive = false;
    swipePointerId = -1;
    if (shouldClose) close();
    else swipeOffset = 0;
  }
</script>

{#if open}
  <div
    class="filtros-mobile-backdrop"
    transition:fade={{ duration: 180 }}
    on:click={close}
    role="presentation"
  ></div>
  <div
    class="filtros-mobile-sheet"
    class:filtros-mobile-sheet--dragging={swipeActive}
    style:transform={`translateY(${swipeOffset}px)`}
    style:--mobile-viewport-height={`${viewportHeight || 800}px`}
    style:--mobile-viewport-bottom={`${viewportBottom}px`}
    role="dialog"
    aria-modal="true"
    aria-labelledby="filtros-mobile-title"
    aria-describedby="filtros-mobile-description"
    use:sheetOverlay={{ onClose: close, returnFocus }}
  >
    <div
      class="filtros-mobile-header"
      data-swipe-region
      role="group"
      aria-label="Encabezado de filtros; desliza hacia abajo para cerrar"
      on:pointerdown={iniciarSwipe}
      on:pointermove={moverSwipe}
      on:pointerup={terminarSwipe}
      on:pointercancel={terminarSwipe}
    >
      <div class="filtros-mobile-handle" aria-hidden="true"></div>
      <div class="filtros-mobile-heading-row">
        <span class="filtros-mobile-title" id="filtros-mobile-title">Filtros</span>
        <span class="sr-only" id="filtros-mobile-description">Los cambios se aplican al catálogo inmediatamente.</span>
        <button class="filtros-mobile-close" data-overlay-initial-focus on:click={close} aria-label="Cerrar">
          <svg width="18" height="18" viewBox="0 0 18 18" fill="none" aria-hidden="true">
            <line x1="3" y1="3" x2="15" y2="15" stroke="currentColor" stroke-width="1.8" stroke-linecap="round"/>
            <line x1="15" y1="3" x2="3" y2="15" stroke="currentColor" stroke-width="1.8" stroke-linecap="round"/>
          </svg>
        </button>
      </div>
      {#if pills.length}
        <div class="filtros-mobile-pills" aria-label="Todos los filtros activos">
          {#each pills as pill (pill.id)}
            <span class="sheet-pill">
              <span>{pill.label}</span>
              <button type="button" aria-label={pill.removeLabel} on:click={() => dispatch('remove', pill)}>×</button>
            </span>
          {/each}
        </div>
      {/if}
    </div>
    <div class="filtros-mobile-body">
      <Filtros
        {lanzamientos}
        {valores}
        on:filtrar
      />
      <details class="sheet-radar">
        <summary>Radar</summary>
        <RadarPanel
          radar={valores.radar}
          bind:this={radarRef}
          on:filtrarRadar
        />
      </details>
      <ResetFiltersButton on:click={() => dispatch('reset')} />
    </div>
  </div>
{/if}
