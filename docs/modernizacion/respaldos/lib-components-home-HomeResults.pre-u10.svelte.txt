<script>
  // Resultados del inicio (E13): carga, error, barra de orden y paginación,
  // grilla con el separador de aproximados, contador y paginación inferior.
  import { createEventDispatcher } from 'svelte';
  import LanzamientoCard from '$lib/components/LanzamientoCard.svelte';
  import Pagination from '$lib/components/Pagination.svelte';
  import SearchResultsBar from '$lib/components/SearchResultsBar.svelte';
  import { RELEASES_PER_PAGE } from '$lib/domain/homeSearchView';

  export let loading = true;
  export let error = false;
  export let announcement = '';
  export let page = 1;
  export let totalPages = 1;
  export let order;
  export let orders = [];
  export let approximateNotice = '';
  /** Todas las coincidencias, exactas y aproximadas. */
  export let matches = [];
  /** Las de la página actual. */
  export let pageMatches = [];
  export let onlyApproximate = false;
  export let separatorAtStart = false;
  export let gridEl = null;

  const dispatch = createEventDispatcher();

  function s(n) { return n === 1 ? '' : 's'; }
</script>

<section class="resultados l-container l-container--wide">
  <p class="sr-only" aria-live="polite" aria-atomic="true">{announcement}</p>

{#if loading}
  <div class="loader" role="status" aria-label="Cargando catálogo">
    <span class="loader__dot"></span>
    <span class="loader__dot"></span>
    <span class="loader__dot"></span>
    <span class="loader__label">Cargando lanzamientos</span>
  </div>
{:else if error}
  <p class="resultados__error" role="alert">No pudimos cargar los lanzamientos. Inténtalo de nuevo.</p>
{:else}
  <SearchResultsBar
    {page}
    {totalPages}
    {order}
    {orders}
    {approximateNotice}
    on:page
    on:order
  />

  {#if !matches.length}
    <p class="resultados__vacio">No encontramos lanzamientos con estos filtros.</p>
  {:else}
    <div class="resultados__grid lanzamientos-grilla" bind:this={gridEl} tabindex="-1">
      {#each pageMatches as coincidencia, index (coincidencia.release.id || coincidencia.release.slug)}
        {#if coincidencia.group === 'approximate' && ((index === 0 && separatorAtStart) || (index > 0 && pageMatches[index - 1].group === 'exact'))}
          <p class="resultados__grupo" aria-label="Resultados aproximados">Resultados aproximados</p>
        {/if}
        <LanzamientoCard lanzamiento={coincidencia.release} onabrir={(release) => dispatch('abrir', release)} />
      {/each}
    </div>

    {#if !onlyApproximate}
      <p class="resultados__contador">
        {#if totalPages > 1}
          {(page - 1) * RELEASES_PER_PAGE + 1}–{Math.min(page * RELEASES_PER_PAGE, matches.length)} de {matches.length} lanzamiento{s(matches.length)}
        {:else}
          {matches.length} lanzamiento{s(matches.length)}
        {/if}
      </p>
    {/if}

    <div class="resultados__paginacion-inferior">
      <Pagination {page} {totalPages} label="Paginación inferior de resultados" on:change={(event) => dispatch('page', event.detail)} />
    </div>
  {/if}
{/if}
</section>
