<script>
  // Panel de búsqueda inline de desktop (U10): una sola caja con el formulario
  // y el Radar lado a lado, y «Reiniciar filtros» al pie como acción secundaria.
  import { createEventDispatcher } from 'svelte';
  import { cubicOut } from 'svelte/easing';
  import Filtros from '$lib/components/Filtros.svelte';
  import RadarPanel from './RadarPanel.svelte';
  import ResetFiltersButton from './ResetFiltersButton.svelte';

  export let open = false;
  export let lanzamientos = [];
  export let valores;
  export let radarRef = null;

  const dispatch = createEventDispatcher();

  function panelIn(node, { duration = 200 } = {}) {
    return {
      duration,
      easing: cubicOut,
      css: (t) => `opacity: ${t}; transform: translateY(${(1 - t) * 8}px);`
    };
  }

  function panelOut(node, { duration = 180 } = {}) {
    const h = node.scrollHeight;
    return {
      duration,
      easing: cubicOut,
      css: (t) => `opacity: ${t}; overflow: hidden; height: ${h * t}px;`
    };
  }
</script>

{#if open}
<section class="filtros-wrap l-container l-container--wide" aria-label="Búsqueda y filtros" in:panelIn out:panelOut>
  <div class="filtros-caja">
    <div class="filtros-grid">
      <div class="filtros-col filtros-col--form">
        <Filtros
          {lanzamientos}
          {valores}
          on:filtrar
        />
      </div>

      <div class="filtros-col filtros-col--radar">
        <RadarPanel
          radar={valores.radar}
          bind:this={radarRef}
          on:filtrarRadar
        />
      </div>
    </div>

    <div class="filtros-pie">
      <ResetFiltersButton on:click={() => dispatch('reset')} />
    </div>
  </div>
</section>
{/if}
