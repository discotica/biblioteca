<script lang="ts">
  import { onDestroy } from 'svelte';
  import { goto } from '$app/navigation';
  import { page } from '$app/stores';
  import AddToCrateButton from './AddToCrateButton.svelte';
  import CoverFallback from './CoverFallback.svelte';
  import RadarMini from './RadarMini.svelte';
  import ReleasePlayer from './ReleasePlayer.svelte';
  import type { ArtistCredit, Release } from '$lib/domain/releases';
  import { summarizeReleaseCredits, VARIOUS_ARTISTS_LABEL } from '$lib/domain/releases';
  import { canonicalReleasePath, formatReleaseDateEsCo } from '$lib/domain/releaseDetail';
  import { createSearchState, parseSearchState, serializeSearchState } from '$lib/domain/searchState';
  import { coverFailure } from '$lib/ui/coverFallback';
  import { dropdownMenu } from '$lib/ui/overlay';
  import { slugify } from '$lib/utils/slugify';
  import { removeBetaInvitationParams } from '$lib/domain/betaInvitation';

  export let release: Release;
  export let mode: 'modal' | 'page' = 'page';
  export let titleId = '';

  let copied = false;
  let copiedTimer: ReturnType<typeof setTimeout> | undefined;
  let menuLayer: HTMLElement;
  let artistsOpen = false;
  let artistsTrigger: HTMLButtonElement;
  let coverFailed = false;

  $: resolvedTitleId = titleId || `release-title-${release.slug}-${mode}`;
  $: coverUrl = release.coverUrl;
  $: coverUrl, (coverFailed = false);
  $: presentation = summarizeReleaseCredits(release);
  $: date = formatReleaseDateEsCo(release.releaseDate);
  $: hasRadar = Object.values(release.radar).some((value) => value > 0);
  $: sharePath = canonicalReleasePath(release);

  function artistHref(credit: ArtistCredit) {
    return credit.slug ? `/artista/${encodeURIComponent(credit.slug)}` : null;
  }

  function closeArtists() {
    artistsOpen = false;
  }

  function filterHref(kind: 'country' | 'genre', value: string) {
    const parsed = parseSearchState($page.url.searchParams);
    const state = createSearchState({
      ...parsed.state,
      [kind === 'country' ? 'countries' : 'genres']: [value],
      modal: null,
      page: 1
    });
    const query = serializeSearchState(state, removeBetaInvitationParams(parsed.unknown)).toString();
    return query ? `/?${query}` : '/';
  }

  function navigateToFilter(kind: 'country' | 'genre', value: string) {
    void goto(filterHref(kind, value), { invalidateAll: true, noScroll: false });
  }

  function share() {
    navigator.clipboard?.writeText?.(`https://biblioteca.discotica.com${sharePath}`)
      ?.then(() => {
        copied = true;
        clearTimeout(copiedTimer);
        copiedTimer = setTimeout(() => (copied = false), 1500);
      })
      .catch(() => {});
  }

  onDestroy(() => clearTimeout(copiedTimer));
</script>

<article
  class="release-detail release-detail--{mode}"
  data-release-detail
  data-release-detail-mode={mode}
  data-release-slug={release.slug}
>
  <div class="release-detail__menu-layer" data-release-menu-layer bind:this={menuLayer}></div>

  <div class="release-detail__layout" data-release-scroll-region>
    <div class="release-detail__cover-column">
      <div class="release-detail__cover">
        {#if coverUrl && !coverFailed}
          <img src={coverUrl} alt={`Portada de ${release.title}`} loading="eager" use:coverFailure={() => (coverFailed = true)} />
        {:else}
          <div class="release-detail__cover-placeholder" role="img" aria-label={`Portada no disponible para ${release.title}`}><CoverFallback /></div>
        {/if}

        {#if hasRadar}
          <div class="release-detail__radar" aria-hidden="true">
            <div class="release-detail__radar-scale"><RadarMini valores={release.radar} /></div>
          </div>
        {/if}
      </div>
    </div>

    <div class="release-detail__info" data-release-scroll-region>
      {#if presentation.label}
        <div class="release-detail__presentation" aria-label="Artistas">
          {#if presentation.various}
            <button
              class="release-detail__artists-trigger"
              type="button"
              bind:this={artistsTrigger}
              aria-haspopup="menu"
              aria-expanded={artistsOpen}
              aria-controls={artistsOpen ? `release-artists-${release.slug}-${mode}` : undefined}
              on:click={() => (artistsOpen = !artistsOpen)}
            >
              {VARIOUS_ARTISTS_LABEL} <span aria-hidden="true">▾</span>
            </button>

            {#if artistsOpen}
              <div
                class="release-detail__artists-menu"
                id={`release-artists-${release.slug}-${mode}`}
                role="menu"
                aria-label="Artistas del lanzamiento"
                tabindex="-1"
                use:dropdownMenu={{ onClose: closeArtists, trigger: artistsTrigger }}
              >
                {#each presentation.credits as credit (`artist-${credit.id ?? credit.slug ?? credit.position}-${credit.name}`)}
                  {#if artistHref(credit)}
                    <a role="menuitem" href={artistHref(credit) ?? undefined}>{credit.name}</a>
                  {:else}
                    <span class="release-detail__artist-unlinked">{credit.name}</span>
                  {/if}
                {/each}
              </div>
            {/if}
          {:else}
            {#each presentation.names as credit, index (`name-${credit.id ?? credit.slug ?? credit.position}-${credit.name}`)}
              {#if artistHref(credit)}
                <a class="release-detail__artist-link" href={artistHref(credit) ?? undefined}>{credit.name}</a>
              {:else}
                <span>{credit.name}</span>
              {/if}{#if index < presentation.names.length - 1}<span class="release-detail__artist-sep" aria-hidden="true">,</span>{/if}
            {/each}
            {#if presentation.moreLabel}<span class="release-detail__artist-more">{presentation.moreLabel}</span>{/if}
          {/if}
        </div>
      {/if}

      <svelte:element this={mode === 'page' ? 'h1' : 'h2'} class="release-detail__title" id={resolvedTitleId} tabindex="-1">
        {release.title}
      </svelte:element>

      <dl class="release-detail__metadata">
        {#if release.releaseType}
          <div><dt>Tipo</dt><dd>{release.releaseType}</dd></div>
        {/if}

        {#if release.countries.length}
          <div>
            <dt>País</dt>
            <dd>
              {#each release.countries as country, index (country)}
                <button type="button" on:click={() => navigateToFilter('country', country)}>{country}</button>{#if index < release.countries.length - 1}<span aria-hidden="true"> / </span>{/if}
              {/each}
            </dd>
          </div>
        {/if}

        {#if release.label}
          <div>
            <dt>Sello</dt>
            <dd><a href={`/sello/${slugify(release.label)}`}>{release.label}</a></dd>
          </div>
        {/if}

        {#if date}
          <div><dt>Fecha</dt><dd>{date}</dd></div>
        {/if}

        {#if release.genres.length}
          <div class="release-detail__metadata-row--genres">
            <dt>Género</dt>
            <dd class="release-detail__tags">
              {#each release.genres as genre (genre)}
                <button class="release-detail__tag" type="button" on:click={() => navigateToFilter('genre', genre)}>{genre}</button>
              {/each}
            </dd>
          </div>
        {/if}
      </dl>

      {#if release.review}
        <blockquote class="release-detail__review">“{release.review}”</blockquote>
      {/if}

      {#key release.slug}
        <ReleasePlayer {release} />
      {/key}

      <div class="release-detail__actions" aria-label="Acciones del lanzamiento">
        {#if release.buyLink}
          <a class="release-detail__buy" href={release.buyLink} target="_blank" rel="noopener noreferrer">Comprar</a>
        {/if}

        <AddToCrateButton releaseId={release.id ?? undefined} lanzamientoSlug={release.slug} menuPortalTarget={menuLayer} />

        <a class="release-detail__share" href={`/reportar${release.id ? `?release_id=${encodeURIComponent(release.id)}` : ''}`}>Reportar</a>

        <button class="release-detail__share" type="button" on:click={share} aria-label="Copiar enlace canónico">
          <svg width="13" height="13" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true">
            <circle cx="18" cy="5" r="3"/><circle cx="6" cy="12" r="3"/><circle cx="18" cy="19" r="3"/>
            <line x1="8.59" y1="13.51" x2="15.42" y2="17.49"/><line x1="15.41" y1="6.51" x2="8.59" y2="10.49"/>
          </svg>
          <span class:release-detail__share-label--ok={copied}>{copied ? 'Copiado' : 'Compartir'}</span>
        </button>
      </div>

      <section class="release-detail__recommendations" aria-label="Recomendaciones" data-release-recommendations>
        <slot name="recommendations"></slot>
      </section>
    </div>
  </div>
</article>
