<script>
  import { createEventDispatcher } from 'svelte';
  import Pagination from '$lib/components/Pagination.svelte';

  export let page = 1;
  export let totalPages = 1;
  export let order = 'fecha-desc';
  export let orders = [];
  export let approximateNotice = '';

  const dispatch = createEventDispatcher();

  function selectOrder(event) {
    dispatch('order', event.currentTarget.value);
  }
</script>

<div class="results-bar">
  <div class="results-bar__main">
    <Pagination {page} {totalPages} label="Paginación superior de resultados" on:change={(event) => dispatch('page', event.detail)} />

    <label class="results-bar__order">
      <span>Ordenar por</span>
      <select value={order} on:change={selectOrder} aria-label="Ordenar resultados">
        {#each orders as option}
          <option value={option.value}>{option.label}</option>
        {/each}
      </select>
    </label>
  </div>

  {#if approximateNotice}
    <p class="results-bar__notice">{approximateNotice}</p>
  {/if}
</div>
