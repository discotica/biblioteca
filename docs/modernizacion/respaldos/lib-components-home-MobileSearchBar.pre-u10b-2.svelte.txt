<script>
  // Línea 1 del inicio en móvil (E15, U10): «Buscar» y el selector de vista.
  // Es la única línea sticky; su visibilidad la decide mobileSearchChrome.
  import { createEventDispatcher } from 'svelte';
  import Icon from '$lib/ui/Icon.svelte';
  import ViewToggle from '$lib/ui/ViewToggle.svelte';

  export let visible = true;
  export let expanded = false;
  export let count = 0;
  export let buttonEl = null;
  /** 'grid' | 'list' */
  export let view = 'grid';

  const dispatch = createEventDispatcher();
</script>

<div
  class="mobile-search-anchor l-container l-container--wide"
  class:mobile-search-anchor--hidden={!visible}
  data-search-visible={visible}
>
  <div class="mobile-search-row">
    <button
      class="mobile-search-bar"
      type="button"
      bind:this={buttonEl}
      aria-label="Abrir filtros de búsqueda — Buscar y filtrar"
      aria-haspopup="dialog"
      aria-expanded={expanded}
      on:click={(event) => dispatch('open', event.currentTarget)}
    >
      <Icon name="buscar" size="md" weight="bold" />
      <span>Buscar</span>
      {#if count > 0}
        <span class="mobile-search-badge" aria-label={`${count} filtros activos`}>{count}</span>
      {/if}
    </button>
    <ViewToggle value={view} label="Vista de resultados" onchange={(mode) => dispatch('view', mode)} />
  </div>
</div>
