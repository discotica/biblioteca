async function mockCanonicalCatalog(page: Page) {
  await page.route('**/rest/v1/rpc/resolve_release_slug', async (route) => {
    const request = route.request();
    const body = request.postDataJSON() as { p_slug?: string };
    const requested = body.p_slug ?? '';
    const canonicalSlug = requested === 'disco-bandcamp-legado' ? 'disco-bandcamp-artista-bc' : requested;
    const row = rows.find((candidate) => candidate.slug === canonicalSlug);
    const response = row
      ? [{
          release_id: row.id,
          canonical_slug: row.slug,
          match_kind: requested === row.slug ? 'canonical' : 'alias',
          alias_kind: requested === row.slug ? null : 'migration'
        }]
      : [];
    await route.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify(response) });
  });

  await page.route('**/rest/v1/catalog_releases**', async (route) => {
    const url = new URL(route.request().url());
    const idFilter = url.searchParams.get('id');
    if (idFilter?.startsWith('eq.')) {
      const row = rows.find((candidate) => candidate.id === idFilter.slice(3));
      await route.fulfill({ status: 200, contentType: 'application/vnd.pgrst.object+json', body: JSON.stringify(row ?? null) });
      return;
    }
    await route.fulfill({
      status: 200,
      contentType: 'application/json',
      headers: { 'content-range': `0-${rows.length - 1}/${rows.length}` },
      body: JSON.stringify(rows)
    });
  });
}

test('E04: el diálogo etiqueta, contiene el foco, bloquea el fondo y lo restaura', async ({ page }) => {
  await page.setViewportSize({ width: 1280, height: 500 });
  await page.goto('/');
  const trigger = page.getByRole('link', { name: 'Abrir Disco Bandcamp' }).first();
  await page.evaluate(() => window.scrollTo(0, 180));
  await trigger.click();

  const dialog = page.getByRole('dialog', { name: 'Disco Bandcamp' });
  const close = dialog.getByRole('button', { name: 'Cerrar' });
  await expect(dialog).toHaveAttribute('aria-modal', 'true');
  await expect(close).toBeFocused();
  await expect(page.locator('header.site-header')).toHaveAttribute('aria-hidden', 'true');
  await expect(page.locator('body')).toHaveCSS('position', 'fixed');
  const scrollAtOpen = await page.locator('body').evaluate((body) => Math.abs(parseInt(body.style.top, 10)));

  await page.keyboard.press('Shift+Tab');
  await expect(dialog.locator(':focus')).toHaveCount(1);
  await page.keyboard.press('Escape');

  await expect(dialog).toHaveCount(0);
  await expect(trigger).toBeFocused();
  await expect(page.locator('body')).not.toHaveCSS('position', 'fixed');
  await expect.poll(() => page.evaluate(() => window.scrollY)).toBe(scrollAtOpen);
});

test('E04: la pila mantiene el bloqueo hasta cerrar la última capa y devuelve el foco correcto', async ({ page }) => {
  await page.goto('/');
  await page.evaluate(async () => {
    const { dialogOverlay } = await import('/src/lib/ui/overlay.ts');
    const opener = document.createElement('button');
    opener.textContent = 'Abrir pila de prueba';
    document.body.append(opener);
    opener.focus();

    const outer = document.createElement('div');
    outer.setAttribute('role', 'dialog');
    outer.setAttribute('aria-modal', 'true');
    outer.dataset.e04Outer = '';
    const outerInitial = document.createElement('button');
    outerInitial.dataset.overlayInitialFocus = '';
    outerInitial.textContent = 'Cerrar capa exterior';
    const innerOpener = document.createElement('button');
    innerOpener.textContent = 'Abrir capa interior';
    outer.append(outerInitial, innerOpener);
    document.body.append(outer);

    let outerAction: ReturnType<typeof dialogOverlay>;
    outerAction = dialogOverlay(outer, {
      onClose: () => {
        outerAction.destroy();
        outer.remove();
      },
      returnFocus: opener
    });

    innerOpener.focus();
    const inner = document.createElement('div');
    inner.setAttribute('role', 'dialog');
    inner.setAttribute('aria-modal', 'true');
    inner.dataset.e04Inner = '';
    const innerInitial = document.createElement('button');
    innerInitial.dataset.overlayInitialFocus = '';
    innerInitial.textContent = 'Cerrar capa interior';
    inner.append(innerInitial);
    document.body.append(inner);

    let innerAction: ReturnType<typeof dialogOverlay>;
    innerAction = dialogOverlay(inner, {
      onClose: () => {
        innerAction.destroy();
        inner.remove();
      },
      returnFocus: innerOpener
    });
  });

  const outer = page.locator('[data-e04-outer]');
  const inner = page.locator('[data-e04-inner]');
  await expect(inner.getByRole('button', { name: 'Cerrar capa interior' })).toBeFocused();
  await expect(outer).toHaveAttribute('aria-hidden', 'true');
  await expect(page.locator('body')).toHaveCSS('position', 'fixed');

  await page.keyboard.press('Escape');
  await expect(inner).toHaveCount(0);
  await expect(outer).not.toHaveAttribute('aria-hidden', 'true');
  await expect(outer.getByRole('button', { name: 'Abrir capa interior' })).toBeFocused();
  await expect(page.locator('body')).toHaveCSS('position', 'fixed');

  await page.keyboard.press('Escape');
  await expect(outer).toHaveCount(0);
  await expect(page.getByRole('button', { name: 'Abrir pila de prueba' })).toBeFocused();
  await expect(page.locator('body')).not.toHaveCSS('position', 'fixed');
});

test('E14: en móvil se oculta al bajar, vuelve solo al inicio y se congela para menú/campo', async ({ page }) => {
  await page.setViewportSize({ width: 390, height: 500 });
  await page.goto('/');
  await page.locator('main').evaluate((main) => {
    const spacer = document.createElement('div');
    spacer.style.height = '1400px';
    const input = document.createElement('input');
    input.dataset.e14FocusField = '';
    spacer.append(input);
    main.append(spacer);
  });
  const header = page.locator('header.site-header');

  await page.evaluate(() => window.scrollTo(0, 120));
  await expect.poll(() => page.evaluate(() => window.scrollY)).toBe(120);
  await expect(header).toHaveAttribute('data-header-visible', 'false');
  await page.evaluate(() => window.scrollTo(0, 32));
  await expect(header).toHaveAttribute('data-header-visible', 'false');
  await page.evaluate(() => window.scrollTo(0, 0));
  await expect(header).toHaveAttribute('data-header-visible', 'true');

  await header.getByRole('button', { name: 'Abrir menú' }).click();
  await page.evaluate(() => window.scrollTo(0, 180));
  await expect(header).toHaveAttribute('data-header-visible', 'true');
  await page.keyboard.press('Escape');

  const field = page.locator('[data-e14-focus-field]');
  await field.focus();
  await page.evaluate(() => window.scrollTo(0, 240));
  await expect(header).toHaveAttribute('data-header-visible', 'true');
  await field.blur();
  await page.waitForTimeout(20);
  await page.evaluate(() => window.scrollTo(0, 280));
  await expect(header).toHaveAttribute('data-header-visible', 'false');
});

  await page.setViewportSize({ width: 1440, height: 900 });
  await authenticateHeaderFixture(page);
  await page.goto('/');

  const header = page.locator('header.site-header');
  const account = header.getByRole('button', { name: 'Cuenta @persona' });
  await expect(account).toBeVisible();
  await expect(header.getByRole('link', { name: 'Entrar' })).toHaveCount(0);
  await expect(header.getByRole('link', { name: 'Perfil' })).toHaveCount(0);
