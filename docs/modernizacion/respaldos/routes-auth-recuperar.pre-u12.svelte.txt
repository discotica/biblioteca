<script lang="ts">
  import AccountLayout from '$lib/layouts/AccountLayout.svelte';
  import { supabase } from '$lib/supabase';
  let email = '';
  let enviado = false;
  let cargando = false;
  async function recuperar() {
    cargando = true;
    await supabase.auth.resetPasswordForEmail(email, {
      redirectTo: `${window.location.origin}/auth/reset?flow=recovery`
    });
    cargando = false;
    enviado = true;
  }
</script>
<svelte:head>
  <title>Recuperar contraseña — Biblioteca Discótica</title>
</svelte:head>
<AccountLayout
  title="Recuperar contraseña"
  help={enviado ? undefined : 'Ingresa tu correo electrónico y te enviaremos un enlace para restablecer tu contraseña.'}
>
    {#if enviado}
      <div class="exito-msg">
        <strong>Listo.</strong>
        <p>Si ese email tiene una cuenta, recibirás un link para cambiar tu contraseña en los próximos minutos.</p>
      </div>
    {:else}
      <form on:submit|preventDefault={recuperar} class="auth-form">
        <label class="campo">
          <span>Email</span>
          <input
            type="email"
            bind:value={email}
            autocomplete="email"
            required
            placeholder="tu@correo.com"
          />
        </label>
        <button type="submit" class="btn-primary" disabled={cargando}>
          {cargando ? 'Enviando…' : 'Enviar link'}
        </button>
      </form>
    {/if}
    {#snippet footer()}
      <a href="/auth/login">Volver a entrar</a>
    {/snippet}
</AccountLayout>
