<script lang="ts">
  import { onMount } from 'svelte';
  import RadarMini from '$lib/components/RadarMini.svelte';
  import { slugifyReleaseIdentity } from '$lib/domain/releases';
  import { user } from '$lib/stores/authStore';
  import { dialogOverlay } from '$lib/ui/overlay';
  import { BELOW_MOBILE_QUERY } from '$lib/ui/breakpoints';
  import {
    blankCatalog,
    createArtist,
    createGenre,
    createLabel,
    createCatalog,
    dictionaries,
    getAdminRole,
    getArtistsByIds,
    getCatalogHistory,
    getCatalogRelease,
    listCatalog,
    revertCatalog,
    saveCatalog,
    searchArtists,
    type Artist,
    type CatalogHistory,
    type CatalogSnapshot,
    type CatalogStatus,
    type DictionaryEntry
  } from '$lib/services/adminCatalog';

  type Tab = 'edit' | 'history';
  type DialogKind = 'publish' | 'archive' | 'revert' | 'artist' | 'label' | 'genre' | null;
  type CreditKind = 'artist_ids' | 'collaborator_ids';
  type CreditEditor = { kind: CreditKind; replaceId?: string } | null;

  const enabled = import.meta.env.VITE_ADMIN_CATALOG_ENABLED === 'true';
  const radarKeys = ['experimental', 'denso', 'sintetico', 'bailable', 'rapido'] as const;
  const statusLabels: Record<CatalogStatus, string> = {
    draft: 'Borrador',
    published: 'Publicado',
    archived: 'Archivado'
  };
  const creditGroups: Array<{ kind: CreditKind; title: string; required: boolean }> = [
    { kind: 'artist_ids', title: 'Artistas principales', required: true },
    { kind: 'collaborator_ids', title: 'Colaboradores', required: false }
  ];

  let role = '';
  let ready = false;
  let busy = false;
  let loading = false;
  let mobile = false;
  let error = '';
  let notice = '';
  let query = '';
  let page = 1;
  let total = 0;
  let filtersOpen = false;
  let statuses: CatalogStatus[] = [];
  let typeIds: string[] = [];
  let searchTimer: ReturnType<typeof setTimeout> | undefined;
  let items: CatalogSnapshot[] = [];
  let checkedIds: string[] = [];
  let selected: CatalogSnapshot | null = null;
  let labelText = '';
  let history: CatalogHistory[] = [];
  let activeTab: Tab = 'edit';
  let observation = '';
  let dialog: DialogKind = null;
  let pendingHistory: CatalogHistory | null = null;
  let artistName = '';
  let dictionaryName = '';
  let dictionaryReason = '';
  let artistQuery = '';
  let artistResults: Artist[] = [];
  let artistSearching = false;
  let artistSearchTimer: ReturnType<typeof setTimeout> | undefined;
  let creditEditor: CreditEditor = null;
  let selectedCreditIds: string[] = [];
  let draggedCreditIds: string[] = [];
  let lastCreditSelection: { kind: CreditKind; index: number } | null = null;
  let artists: Artist[] = [];
  let labels: DictionaryEntry[] = [];
  let genres: DictionaryEntry[] = [];
  let countries: DictionaryEntry[] = [];
  let types: DictionaryEntry[] = [];

  const isAdmin = () => enabled && role === 'admin';
  const toggle = <T,>(values: T[], value: T) =>
    values.includes(value) ? values.filter((current) => current !== value) : [...values, value];
  const artistOf = (id: string) => artists.find((artist) => artist.id === id);
  const nameOf = (entries: DictionaryEntry[], id: string | null) => entries.find((entry) => entry.id === id)?.name ?? 'Sin asignar';
  const selectedArtistNames = (ids: string[]) => ids.map((id) => artistOf(id)?.name ?? 'Artista sin resolver').join(', ');

  function normalize(value: string) {
    return value.trim().toLocaleLowerCase('es').normalize('NFD').replace(/\p{Diacritic}/gu, '').replace(/\s+/g, ' ');
  }
  function formatDate(value: string | null, includeTime = false) {
    if (!value) return 'Sin fecha';
    const parsed = new Date(value.length === 10 ? `${value}T12:00:00` : value);
    return new Intl.DateTimeFormat('es-CO', includeTime ? { dateStyle: 'medium', timeStyle: 'short' } : { day: '2-digit', month: 'short', year: 'numeric' }).format(parsed);
  }
  function primaryLink(snapshot = selected) {
    return snapshot?.links.find((link) => link.is_primary) ?? null;
  }
  function inferProvider(url: string) {
    try {
      const host = new URL(url).hostname.toLowerCase();
      if (host.includes('bandcamp.com')) return 'Bandcamp';
      if (host.includes('spotify.com')) return 'Spotify';
      if (host.includes('soundcloud.com')) return 'SoundCloud';
    } catch {
      // La validación muestra el error cuando la persona termina de editar.
    }
    return 'Enlace';
  }
  function missingFields(snapshot: CatalogSnapshot) {
    const missing: string[] = [];
    if (!snapshot.title.trim()) missing.push('Título');
    if (!snapshot.slug.trim()) missing.push('Identificador interno');
    if (!snapshot.release_date) missing.push('Fecha');
    if (!snapshot.release_type_id) missing.push('Tipo');
    if (!snapshot.label_id) missing.push('Sello');
    if (!snapshot.artist_ids.length) missing.push('Artista principal');
    if (!snapshot.genre_ids.length) missing.push('Género');
    if (!snapshot.country_ids.length) missing.push('País');
    if (!snapshot.cover_url || !/^https:\/\//i.test(snapshot.cover_url)) missing.push('Portada');
    const primaryLinks = snapshot.links.filter((link) => link.is_primary);
    if (primaryLinks.length !== 1 || !primaryLinks[0].provider.trim() || !/^https:\/\//i.test(primaryLinks[0].url)) missing.push('Link de compra');
    return missing;
  }
  function completion(snapshot: CatalogSnapshot) {
    return Math.round(((10 - missingFields(snapshot).length) / 10) * 100);
  }
  function validate(snapshot: CatalogSnapshot, publishing = false) {
    if (!snapshot.title.trim()) return 'El título es obligatorio.';
    if (!/^[a-z0-9]+(?:-[a-z0-9]+)*(?:--[a-z0-9]+)?$/.test(snapshot.slug)) return 'No pudimos generar el identificador interno. Revisa el título.';
    if (!snapshot.artist_ids.length) return 'Selecciona al menos un artista principal.';
    if (snapshot.artist_ids.some((id) => snapshot.collaborator_ids.includes(id))) return 'Un artista no puede ser principal y colaborador a la vez.';
    if (publishing && missingFields(snapshot).length) return `Completa antes de publicar: ${missingFields(snapshot).join(', ')}.`;
    if (snapshot.cover_url && !/^https:\/\//i.test(snapshot.cover_url)) return 'La portada debe usar HTTPS.';
    if (snapshot.links.some((link) => !link.provider.trim() || !/^https:\/\//i.test(link.url))) return 'Cada enlace requiere proveedor y una URL HTTPS válida.';
    if (snapshot.links.filter((link) => link.is_primary).length > 1) return 'Solo puede haber un enlace principal.';
    return '';
  }
  function assignSnapshot(snapshot: CatalogSnapshot) {
    selected = snapshot;
    labelText = snapshot.label_id ? nameOf(labels, snapshot.label_id) : '';
  }
  async function ensureArtists(ids: string[]) {
    const known = new Set(artists.map((artist) => artist.id));
    const missing = [...new Set(ids)].filter((id) => id && !known.has(id));
    if (missing.length) artists = [...artists, ...(await getArtistsByIds(missing))];
  }
  async function loadList() {
    if (!isAdmin()) return;
    loading = true;
    error = '';
    try {
      const result = await listCatalog({ query, statuses, qualities: [], typeIds, page });
      items = result.items;
      total = result.total;
      await ensureArtists(items.flatMap((item) => [...item.artist_ids, ...item.collaborator_ids]));
    } catch (cause: any) {
      error = cause?.code === '42501' ? 'El servidor denegó el acceso administrativo.' : cause.message || 'No se pudo cargar el catálogo.';
    } finally {
      loading = false;
    }
  }
  function scheduleListSearch() {
    clearTimeout(searchTimer);
    page = 1;
    searchTimer = setTimeout(loadList, 260);
  }
  async function openRelease(id: string) {
    loading = true;
    error = '';
    notice = '';
    try {
      const snapshot = await getCatalogRelease(id);
      history = await getCatalogHistory(id);
      await ensureArtists([...snapshot.artist_ids, ...snapshot.collaborator_ids]);
      assignSnapshot(snapshot);
      activeTab = 'edit';
      observation = '';
      creditEditor = null;
      selectedCreditIds = [];
      window.scrollTo({ top: 0, behavior: 'smooth' });
    } catch (cause: any) {
      error = cause.message || 'No se pudo abrir el lanzamiento.';
    } finally {
      loading = false;
    }
  }
  function closeEditor() {
    selected = null;
    history = [];
    observation = '';
    error = '';
    notice = '';
    creditEditor = null;
    selectedCreditIds = [];
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }
  function newRelease() {
    assignSnapshot(blankCatalog());
    history = [];
    observation = '';
    activeTab = 'edit';
    error = '';
    notice = '';
    creditEditor = null;
    selectedCreditIds = [];
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }
  async function refreshSelected(id = selected?.id) {
    if (!id) return;
    const snapshot = await getCatalogRelease(id);
    history = await getCatalogHistory(id);
    await ensureArtists([...snapshot.artist_ids, ...snapshot.collaborator_ids]);
    assignSnapshot(snapshot);
    await loadList();
  }
  async function submit(operation: 'update' | 'publish' | 'archive' = 'update') {
    if (!selected || mobile) return;
    const invalid = validate(selected, operation === 'publish');
    if (invalid) { error = invalid; return; }
    if (!observation.trim()) { error = 'Añade una observación del cambio. El historial auditado la exige.'; return; }
    busy = true;
    error = '';
    try {
      const wasNew = !selected.id;
      const outcome = selected.id ? await saveCatalog(selected, observation, operation) : await createCatalog(selected, observation);
      await refreshSelected(outcome.release_id);
      notice = operation === 'archive' ? 'Lanzamiento archivado.' : operation === 'publish' ? 'Lanzamiento publicado.' : wasNew ? 'Borrador creado.' : 'Cambios guardados.';
      observation = '';
    } catch (cause: any) {
      if (cause?.code === '40001') {
        await refreshSelected();
        error = 'Otra persona editó este lanzamiento. Cargamos la versión más reciente; revisa el historial antes de guardar.';
      } else error = cause.message || 'No se pudo guardar.';
    } finally {
      busy = false;
      dialog = null;
    }
  }
  async function submitRevert() {
    if (!pendingHistory || !selected || mobile) return;
    if (!observation.trim()) { error = 'Añade una observación para explicar la reversión.'; return; }
    busy = true;
    error = '';
    try {
      await revertCatalog(pendingHistory.change_set_id, pendingHistory.result_version, observation);
      await refreshSelected();
      notice = 'Cambio revertido mediante una nueva entrada auditada.';
      observation = '';
    } catch (cause: any) {
      if (cause?.code === '40001') {
        await refreshSelected();
        error = 'No se revirtió porque existe una edición posterior. Cargamos el estado actual.';
      } else error = cause.message || 'No se pudo revertir.';
    } finally {
      busy = false;
      dialog = null;
      pendingHistory = null;
    }
  }
  function updateTitle(value: string) {
    if (!selected) return;
    selected.title = value;
    if (!selected.id) selected.slug = slugifyReleaseIdentity(value);
    selected = { ...selected };
  }
  function updateLabel(value: string) {
    if (!selected) return;
    labelText = value;
    const match = labels.find((label) => normalize(label.name) === normalize(value));
    selected.label_id = match?.id ?? null;
    selected = { ...selected };
  }
  function updateBuyLink(value: string) {
    if (!selected) return;
    const provider = inferProvider(value);
    const primaryIndex = selected.links.findIndex((link) => link.is_primary);
    if (primaryIndex < 0) {
      if (value) selected.links = [{ provider, url: value, is_primary: true }, ...selected.links];
    } else if (!value) selected.links = selected.links.filter((_, index) => index !== primaryIndex);
    else selected.links = selected.links.map((link, index) => index === primaryIndex ? { provider, url: value, is_primary: true } : link);
    selected = { ...selected };
  }
  function toggleDictionary(key: 'genre_ids' | 'country_ids', id: string) {
    if (!selected || mobile) return;
    selected[key] = toggle(selected[key], id);
    selected = { ...selected };
  }
  function beginCreditEditor(kind: CreditKind, replaceId?: string) {
    creditEditor = { kind, replaceId };
    artistQuery = replaceId ? artistOf(replaceId)?.name ?? '' : '';
    artistResults = [];
    artistName = '';
  }
  function closeCreditEditor() {
    creditEditor = null;
    artistQuery = '';
    artistResults = [];
  }
  async function runArtistSearch() {
    if (artistQuery.trim().length < 2) { artistResults = []; return; }
    artistSearching = true;
    try { artistResults = await searchArtists(artistQuery); }
    catch (cause: any) { error = cause.message || 'No se pudo buscar artistas.'; }
    finally { artistSearching = false; }
  }
  function scheduleArtistSearch() {
    clearTimeout(artistSearchTimer);
    artistSearchTimer = setTimeout(runArtistSearch, 220);
  }
  function chooseArtist(artist: Artist) {
    if (!selected || !creditEditor || mobile) return;
    const { kind, replaceId } = creditEditor;
    const otherKind: CreditKind = kind === 'artist_ids' ? 'collaborator_ids' : 'artist_ids';
    if (selected[otherKind].includes(artist.id) && artist.id !== replaceId) {
      error = `${artist.name} ya está en el otro grupo de créditos.`;
      return;
    }
    if (replaceId) selected[kind] = selected[kind].map((id) => id === replaceId ? artist.id : id).filter((id, index, values) => values.indexOf(id) === index);
    else if (!selected[kind].includes(artist.id)) selected[kind] = [...selected[kind], artist.id];
    if (!artists.some((entry) => entry.id === artist.id)) artists = [...artists, artist];
    selected = { ...selected };
    selectedCreditIds = [];
    closeCreditEditor();
  }
  function removeCredit(kind: CreditKind, id: string) {
    if (!selected || mobile) return;
    selected[kind] = selected[kind].filter((value) => value !== id);
    selected = { ...selected };
    selectedCreditIds = selectedCreditIds.filter((value) => value !== id);
  }
  function moveCredits(ids: string[], targetKind: CreditKind) {
    if (!selected || mobile || !ids.length) return;
    const moving = [...new Set(ids)];
    const target = selected[targetKind].filter((id) => !moving.includes(id));
    selected.artist_ids = selected.artist_ids.filter((id) => !moving.includes(id));
    selected.collaborator_ids = selected.collaborator_ids.filter((id) => !moving.includes(id));
    selected[targetKind] = [...target, ...moving];
    selected = { ...selected };
    selectedCreditIds = [];
    draggedCreditIds = [];
  }
  function selectCredit(event: MouseEvent, kind: CreditKind, id: string, index: number) {
    if (event.shiftKey && lastCreditSelection?.kind === kind && selected) {
      const start = Math.min(lastCreditSelection.index, index);
      const end = Math.max(lastCreditSelection.index, index);
      selectedCreditIds = [...new Set([...selectedCreditIds, ...selected[kind].slice(start, end + 1)])];
    } else if (event.metaKey || event.ctrlKey) selectedCreditIds = toggle(selectedCreditIds, id);
    else selectedCreditIds = [id];
    lastCreditSelection = { kind, index };
  }
  function startCreditDrag(event: DragEvent, kind: CreditKind, id: string, index: number) {
    if (!selectedCreditIds.includes(id)) {
      selectedCreditIds = [id];
      lastCreditSelection = { kind, index };
    }
    draggedCreditIds = [...selectedCreditIds];
    event.dataTransfer?.setData('text/plain', id);
    if (event.dataTransfer) event.dataTransfer.effectAllowed = 'move';
  }
  async function confirmArtist() {
    if (!artistName.trim() || !selected || !creditEditor || mobile) return;
    busy = true;
    error = '';
    try {
      const result = await createArtist(artistName);
      chooseArtist({ id: result.artist_id, name: result.name });
      notice = result.created ? 'Artista creado y añadido al lanzamiento.' : 'Usamos la identidad canónica existente.';
      dialog = null;
      artistName = '';
    } catch (cause: any) { error = cause.message || 'No se pudo crear el artista.'; }
    finally { busy = false; }
  }
  function beginDictionaryCreate(kind: 'label' | 'genre', name = '') {
    if (mobile) return;
    dictionaryName = name;
    dictionaryReason = '';
    dialog = kind;
  }
  async function confirmDictionary() {
    if (!selected || (dialog !== 'label' && dialog !== 'genre')) return;
    if (!dictionaryName.trim()) { error = 'Escribe el nombre que quieres dar de alta.'; return; }
    if (!dictionaryReason.trim()) { error = 'Añade una observación para auditar esta alta.'; return; }
    const kind = dialog;
    busy = true;
    error = '';
    try {
      const result = kind === 'label'
        ? await createLabel(dictionaryName, dictionaryReason)
        : await createGenre(dictionaryName, dictionaryReason);
      if (kind === 'label') {
        labels = [...labels.filter((entry) => entry.id !== result.id), result].sort((a, b) => a.name.localeCompare(b.name, 'es'));
        selected.label_id = result.id;
        labelText = result.name;
      } else {
        genres = [...genres.filter((entry) => entry.id !== result.id), result].sort((a, b) => a.name.localeCompare(b.name, 'es'));
        if (!selected.genre_ids.includes(result.id)) selected.genre_ids = [...selected.genre_ids, result.id];
      }
      selected = { ...selected };
      notice = result.created ? `${kind === 'label' ? 'Sello' : 'Género'} creado y auditado.` : 'Usamos el valor canónico existente.';
      dialog = null;
      dictionaryName = '';
      dictionaryReason = '';
    } catch (cause: any) {
      error = cause?.code === '42501' ? 'El servidor denegó el alta administrativa.' : cause.message || 'No se pudo crear el valor.';
    } finally { busy = false; }
  }
  function diffFields(entry: CatalogHistory) {
    const before = entry.before_state ?? {};
    const after = entry.after_state ?? {};
    return [...new Set([...Object.keys(before), ...Object.keys(after)])].filter((key) => key !== 'catalog_version' && JSON.stringify(before[key]) !== JSON.stringify(after[key]));
  }

  onMount(async () => {
    const media = window.matchMedia(BELOW_MOBILE_QUERY);
    const updateMobile = () => mobile = media.matches;
    updateMobile();
    media.addEventListener('change', updateMobile);
    if (!$user || !enabled) {
      ready = true;
      return () => media.removeEventListener('change', updateMobile);
    }
    try {
      role = await getAdminRole($user.id);
      if (role === 'admin') {
        const values = await dictionaries();
        ({ labels, genres, countries, types } = values);
        await loadList();
      }
    } catch (cause: any) { error = cause.message || 'No se pudo comprobar el acceso.'; }
    finally { ready = true; }
    return () => {
      media.removeEventListener('change', updateMobile);
      clearTimeout(searchTimer);
      clearTimeout(artistSearchTimer);
    };
  });
</script>

<svelte:head><title>Catálogo administrativo — Discótica</title></svelte:head>

<div class="l-container l-container--full console-shell">
  {#if !enabled}
    <div class="gate"><span>Administración</span><h1>Consola desactivada</h1><p>El kill switch del entorno está apagado.</p></div>
  {:else if !$user}
    <div class="gate"><span>Administración</span><h1>Inicia sesión</h1><p>Necesitas una sesión administrativa para consultar esta ruta.</p><a class="button primary" href="/auth/login">Ir a iniciar sesión</a></div>
  {:else if !ready}
    <div class="gate"><div class="spinner"></div><h1>Comprobando acceso…</h1></div>
  {:else if role !== 'admin'}
    <div class="gate" role="alert"><span>Acceso restringido</span><h1>Esta consola es solo para administradores</h1><p>El servidor también denegará cualquier operación forzada.</p></div>
  {:else if selected}
    <header class="editor-topbar">
      <button class="icon-button" on:click={closeEditor} aria-label="Volver al catálogo">←</button>
      <div class="release-identity">
        {#if selected.cover_url}<img src={selected.cover_url} alt="" />{:else}<div class="cover-placeholder">♪</div>{/if}
        <div><p>{selected.id ? 'Editar lanzamiento' : 'Nuevo lanzamiento'}</p><h1>{selected.title || 'Sin título'}</h1><span>{selectedArtistNames(selected.artist_ids) || 'Añade un artista principal'}</span></div>
      </div>
      <div class="topbar-meta"><span class={`status status-${selected.status}`}>{statusLabels[selected.status]}</span>{#if selected.id}<small>v{selected.catalog_version}</small>{/if}</div>
    </header>
    {#if mobile}<div class="mobile-readonly" role="status">Vista de consulta. Abre esta ficha en desktop para editar.</div>{/if}
    {#if error}<div class="flash error" role="alert">{error}<button on:click={() => error = ''} aria-label="Cerrar mensaje">×</button></div>{/if}
    {#if notice}<div class="flash notice" role="status">{notice}<button on:click={() => notice = ''} aria-label="Cerrar mensaje">×</button></div>{/if}

    <nav class="tabs" aria-label="Secciones del lanzamiento">
      <button class:active={activeTab === 'edit'} on:click={() => activeTab = 'edit'}>Edición</button>
      <button class:active={activeTab === 'history'} on:click={() => activeTab = 'history'}>Historial {#if history.length}<span>{history.length}</span>{/if}</button>
    </nav>

    <div class="editor-layout">
      <main class="editor-main">
        {#if activeTab === 'edit'}
          <div class="editor-flow">
            <section class="editor-section essentials-section">
              <header class="section-heading"><div><p>Información</p><h2>Datos esenciales</h2></div><span>Lo necesario para identificar y publicar el lanzamiento.</span></header>
              <div class="field-grid">
                <label class="span-2">Título <span>*</span><input value={selected.title} on:input={(event) => updateTitle(event.currentTarget.value)} disabled={mobile} placeholder="Título del lanzamiento" /></label>
                <label class="span-2 important-field">Link de compra o escucha <span>*</span><input value={primaryLink()?.url ?? ''} on:input={(event) => updateBuyLink(event.currentTarget.value)} disabled={mobile} placeholder="https://bandcamp.com/…" /><small>Detectamos el proveedor a partir de la URL.</small></label>
                <label>Tipo <span>*</span><select bind:value={selected.release_type_id} disabled={mobile}><option value={null}>Selecciona un tipo</option>{#each types as type}<option value={type.id}>{type.name}</option>{/each}</select></label>
                <label>Fecha de lanzamiento <span>*</span><input type="date" bind:value={selected.release_date} disabled={mobile} /></label>
                <label class="span-2">Sello <span>*</span><input list="catalog-labels" value={labelText} on:input={(event) => updateLabel(event.currentTarget.value)} disabled={mobile} placeholder="Busca un sello existente" /><datalist id="catalog-labels">{#each labels as label}<option value={label.name}></option>{/each}</datalist>{#if labelText && !selected.label_id}<small class="field-warning">No coincide con un sello existente.</small>{#if !mobile}<button class="text-button dictionary-create" on:click={() => beginDictionaryCreate('label', labelText)}>Crear sello confirmado</button>{/if}{:else if !mobile}<button class="text-button dictionary-create" on:click={() => beginDictionaryCreate('label')}>Crear sello confirmado</button>{/if}</label>
              </div>
              <details class="optional-block"><summary>Reseña opcional</summary><label>Texto editorial<textarea rows="5" bind:value={selected.review} disabled={mobile} placeholder="Añade una reseña sólo cuando aporte contexto."></textarea></label></details>
            </section>

            <section class="editor-section">
              <header class="section-heading"><div><p>Créditos</p><h2>Artistas del lanzamiento</h2></div><span>Arrastra entre grupos. Usa Shift para seleccionar varios.</span></header>
              <div class="credit-columns">
                {#each creditGroups as group}
                  <section class:drop-target={draggedCreditIds.length > 0} class="credit-group" role="group" aria-label={group.title} on:dragover|preventDefault on:drop|preventDefault={() => moveCredits(draggedCreditIds, group.kind)}>
                    <header><div><h3>{group.title}{#if group.required} <span>*</span>{/if}</h3><small>{selected[group.kind].length} asignados</small></div>{#if !mobile}<button class="button ghost" on:click={() => beginCreditEditor(group.kind)}>＋ Añadir</button>{/if}</header>
                    <div class="credit-list" role="list" aria-label={group.title}>
