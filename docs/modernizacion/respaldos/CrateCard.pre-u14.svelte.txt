<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import type { Crate } from '$lib/stores/cratesStore';
  import { eliminarCrate } from '$lib/stores/cratesStore';
  import { user } from '$lib/stores/authStore';

  export let crate: Crate;
  export let editable = false;

  const dispatch = createEventDispatcher<{ eliminar: { id: string } }>();

  let confirmandoEliminar = false;
  $: username = typeof $user?.user_metadata?.username === 'string'
    ? $user.user_metadata.username
    : '__self__';

  async function confirmarEliminar() {
    await eliminarCrate(crate.id);
    dispatch('eliminar', { id: crate.id });
    confirmandoEliminar = false;
  }
</script>

<article class="crate-card">
  <a class="crate-link" href="/coleccion/{crate.slug}?usuario={encodeURIComponent(username)}">
    <div class="crate-header">
      <span class="crate-badge" class:publico={crate.is_public} class:privado={!crate.is_public}>
        {crate.is_public ? 'Pública' : 'Privada'}
      </span>
    </div>

    <h3 class="crate-nombre">{crate.name}</h3>

    {#if crate.description}
      <p class="crate-desc">{crate.description}</p>
    {/if}

    <span class="crate-conteo">
      {crate.item_count ?? 0} {(crate.item_count ?? 0) === 1 ? 'lanzamiento' : 'lanzamientos'}
    </span>
  </a>

  {#if editable}
    <div class="crate-acciones">
      {#if confirmandoEliminar}
        <span class="confirmar-texto">¿Eliminar?</span>
        <button class="btn-danger" on:click={confirmarEliminar}>Sí, eliminar</button>
        <button class="btn-ghost" on:click={() => (confirmandoEliminar = false)}>Cancelar</button>
      {:else}
        <button class="btn-ghost" on:click={() => (confirmandoEliminar = true)}>
          Eliminar
        </button>
      {/if}
    </div>
  {/if}
</article>
