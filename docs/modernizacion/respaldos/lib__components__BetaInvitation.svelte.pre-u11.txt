<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import { user } from '$lib/stores/authStore';
  import { dialogOverlay } from '$lib/ui/overlay';
  import { BETA_FEEDBACK_EMAIL, BETA_INVITATION_COPY } from '$lib/domain/betaInvitation';

  export let name: string | null = null;
  export let returnFocus: HTMLElement | null = null;

  const dispatch = createEventDispatcher<{ close: void }>();
  const titleId = 'beta-invitation-title';
  const descriptionId = 'beta-invitation-description';
  const reportLoginHref = `/auth/login?redirect=${encodeURIComponent('/reportar')}`;

  function close() {
    dispatch('close');
  }
</script>

<!-- Keyboard handling and focus containment are provided by E04. -->
<!-- svelte-ignore a11y-click-events-have-key-events a11y-no-static-element-interactions -->
<div class="beta-invitation-backdrop" data-beta-invitation on:click|self={close}>
  <div
    class="beta-invitation"
    role="dialog"
    aria-modal="true"
    aria-labelledby={titleId}
    aria-describedby={descriptionId}
    use:dialogOverlay={{ onClose: close, returnFocus }}
  >
    <button class="beta-invitation__close" data-overlay-initial-focus type="button" on:click={close} aria-label="Cerrar invitación beta">
      <span aria-hidden="true">×</span>
    </button>
    <p class="beta-invitation__eyebrow">Discótica</p>
    <h1 id={titleId}>{name ? `Hola, ${name}.` : BETA_INVITATION_COPY.genericTitle}</h1>
    <p id={descriptionId}>{BETA_INVITATION_COPY.body}</p>
    {#if BETA_INVITATION_COPY.launchDate}
      <p class="beta-invitation__date">{BETA_INVITATION_COPY.launchDate}</p>
    {/if}

    <div class="beta-invitation__actions">
      <button class="ui-button ui-button--primary" type="button" on:click={close}>{BETA_INVITATION_COPY.cta}</button>
    </div>

    <div class="beta-invitation__feedback">
      <p>¿Prefieres escribirnos? Puedes contactarnos por correo durante la beta.</p>
      <a href={`mailto:${BETA_FEEDBACK_EMAIL}`}>Escribir a {BETA_FEEDBACK_EMAIL}</a>
      {#if $user}
        <p>Si iniciaste sesión, también puedes usar el canal protegido de la plataforma.</p>
        <a href="/reportar">Reportar un problema</a>
      {:else}
        <p>Para enviar un reporte protegido desde la plataforma, inicia sesión.</p>
        <a href={reportLoginHref}>Iniciar sesión para reportar un problema</a>
      {/if}
    </div>
  </div>
</div>
