<script lang="ts">
  import { onMount } from 'svelte';
  import { fetchReleasesForCrate, fetchReleases, type Release } from '$lib/utils/catalogoStore';
  import { slugFromLanzamiento } from '$lib/utils/slugLanzamiento';
  import { quitarItem } from '$lib/stores/cratesStore';
  import type { CrateItem } from '$lib/stores/cratesStore';
  import LanzamientoCard from '$lib/components/LanzamientoCard.svelte';
  import LanzamientoModal from '$lib/components/LanzamientoModal.svelte';
  import PageLayout from '$lib/layouts/PageLayout.svelte';

  export let data: {
    crate: {
      id: string;
      name: string;
      description: string | null;
      slug: string;
      is_public: boolean;
      owner_id: string;
      profiles: { username: string; display_name: string | null };
    };
    items: CrateItem[];
    esDueno: boolean;
  };

  let lanzamientosPorSlug: Map<string, Release> = new Map();
  let lanzamientosPorId: Map<string, Release> = new Map();
  let cargando = true;
  let copiado = false;
  let items = data.items;
  let lanzamientoActivo: Release | null = null;

  onMount(async () => {
    const todos = await fetchReleases();
    for (const l of todos) {
      if (l.id) lanzamientosPorId.set(l.id, l);
      lanzamientosPorSlug.set(l.slug, l);
      if (l.legacySlug) lanzamientosPorSlug.set(l.legacySlug, l);
    }
    // Los crates ya migrados guardan release_id y slug canónico; el TSV legado
    // no conoce esos slugs con hash, por lo que completamos el mapa desde el
    // catálogo canónico sin cambiar la fuente del resto de la página.
    try {
      const canonicos = await fetchReleasesForCrate(
        items.map((item) => item.release_id ?? ''),
        items.map((item) => item.lanzamiento_slug)
      );
      for (const l of canonicos) {
        if (l.id) lanzamientosPorId.set(l.id, l);
        if (l.slug) lanzamientosPorSlug.set(l.slug, l);
      }
    } catch (error) {
      console.error('No se pudo resolver el catálogo canónico del crate:', error);
    }
    lanzamientosPorSlug = lanzamientosPorSlug;
    lanzamientosPorId = lanzamientosPorId;
    cargando = false;
  });

  function copiarUrl() {
    const url = new URL(window.location.href);
    url.searchParams.set('usuario', owner.username);
    navigator.clipboard?.writeText(url)
      ?.then(() => { copiado = true; setTimeout(() => (copiado = false), 1500); })
      .catch(() => {});
  }

  async function quitar(item: CrateItem) {
    await quitarItem(data.crate.id, item.id);
    items = items.filter((i) => i.id !== item.id);
  }

  function abrirLanzamiento(lanzamiento: Release) {
    lanzamientoActivo = lanzamiento;
  }

  function cerrarLanzamiento() {
    lanzamientoActivo = null;
  }

  $: owner = data.crate.profiles;
  $: titulo = data.crate.name;
  $: migas = data.esDueno
    ? [{ label: 'Mis colecciones', href: '/mis-colecciones' }, { label: titulo }]
    : [
        { label: `@${owner.username}`, href: `/perfil/${owner.username}` },
        { label: 'Colecciones', href: `/perfil/${owner.username}#colecciones` },
        { label: titulo }
      ];
</script>

<svelte:head>
  <title>{titulo} — Biblioteca Discótica</title>
</svelte:head>

<PageLayout breadcrumbs={migas} title={titulo} description={data.crate.description || undefined}>
  {#snippet meta()}
    <div class="crate-info">
      <span class="badge" class:publico={data.crate.is_public} class:privado={!data.crate.is_public}>
        {data.crate.is_public ? 'Pública' : 'Privada'}
      </span>
      <span class="conteo">{items.length} {items.length === 1 ? 'lanzamiento' : 'lanzamientos'}</span>
    </div>
  {/snippet}
  {#snippet actions()}
    {#if data.crate.is_public}
      <button class="btn-compartir" on:click={copiarUrl}>
        {copiado ? 'Copiado ✓' : 'Copiar link'}
      </button>
    {/if}
  {/snippet}

  {#if cargando}
    <p class="estado">Cargando catálogo…</p>
  {:else if items.length === 0}
    <div class="vacio">
      <p>Esta colección está vacía.</p>
      {#if data.esDueno}
        <a href="/" class="link-accion">Explorar el catálogo</a>
      {/if}
    </div>
  {:else}
    <div class="items-grid lanzamientos-grilla">
      {#each items as item (item.id)}
        {@const l = (item.release_id ? lanzamientosPorId.get(item.release_id) : undefined) ?? lanzamientosPorSlug.get(item.lanzamiento_slug)}
        <article class="item-card">
          {#if l}
            <LanzamientoCard lanzamiento={l} onabrir={abrirLanzamiento} />
            {#if item.note}
              <p class="item-nota">“{item.note}”</p>
            {/if}
          {:else}
            <div class="item-no-encontrado">
              <span class="item-titulo item-titulo--roto">[lanzamiento eliminado]</span>
              <code class="item-slug">{item.lanzamiento_slug}</code>
            </div>
          {/if}

          {#if data.esDueno}
            <button class="btn-quitar" on:click={() => quitar(item)} title="Quitar de esta colección" aria-label={`Quitar ${l?.title ?? 'lanzamiento'} de la colección`}>
              ✕
            </button>
          {/if}
        </article>
      {/each}
    </div>
  {/if}
</PageLayout>

{#if lanzamientoActivo}
  {#key slugFromLanzamiento(lanzamientoActivo)}
    <LanzamientoModal lanzamiento={lanzamientoActivo} on:cerrar={cerrarLanzamiento} />
  {/key}
{/if}
