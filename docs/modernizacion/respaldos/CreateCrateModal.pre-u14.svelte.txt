<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import { crearCrate } from '$lib/stores/cratesStore';
  import { dialogOverlay } from '$lib/ui/overlay';

  export let abierto = false;

  const dispatch = createEventDispatcher<{ creado: { id: string }; cerrar: void }>();

  let nombre = '';
  let descripcion = '';
  let esPublico = true;
  let cargando = false;
  let error = '';
  let nombreInput: HTMLInputElement;

  async function crear() {
    error = '';
    if (!nombre.trim()) {
      error = 'El nombre es obligatorio.';
      return;
    }
    cargando = true;
    const nuevo = await crearCrate(nombre.trim(), descripcion.trim(), esPublico);
    cargando = false;
    if (!nuevo) {
      error = 'No se pudo crear la colección. Intenta de nuevo.';
      return;
    }
    nombre = '';
    descripcion = '';
    esPublico = true;
    dispatch('creado', { id: nuevo.id });
    cerrar();
  }

  function cerrar() {
    abierto = false;
    dispatch('cerrar');
  }
</script>

{#if abierto}
  <!-- The pointer target is a modal backdrop; keyboard handling lives on the dialog action. -->
  <!-- svelte-ignore a11y-click-events-have-key-events a11y-no-static-element-interactions -->
  <div class="overlay" on:click|self={cerrar}>
    <div
      class="modal"
      role="dialog"
      aria-modal="true"
      aria-labelledby="create-crate-title"
      aria-describedby="create-crate-description"
      use:dialogOverlay={{ onClose: cerrar }}
    >
      <h2 class="modal-titulo" id="create-crate-title">Nueva colección</h2>
      <p class="sr-only" id="create-crate-description">Crea una colección de lanzamientos y elige su visibilidad.</p>

      <form on:submit|preventDefault={crear} class="modal-form">
        <label class="campo">
          <span>Nombre</span>
          <input
            type="text"
            bind:value={nombre}
            maxlength="120"
            required
            placeholder="Ej: Cumbia acida 2024"
            data-overlay-initial-focus
            bind:this={nombreInput}
          />
        </label>

        <label class="campo">
          <span>Descripción <em class="opcional">(opcional)</em></span>
          <textarea
            bind:value={descripcion}
            rows="3"
            maxlength="400"
            placeholder="¿Qué tiene esta colección?"
          ></textarea>
        </label>

        <div class="visibilidad">
          <span class="visibilidad-label">Visibilidad</span>
          <div class="toggle-group">
            <button
              type="button"
              class="toggle-btn"
              class:activo={esPublico}
              on:click={() => (esPublico = true)}
            >
              Pública
            </button>
            <button
              type="button"
              class="toggle-btn"
              class:activo={!esPublico}
              on:click={() => (esPublico = false)}
            >
              Privada
            </button>
          </div>
          <small class="campo-ayuda">
            {esPublico ? 'Cualquier persona puede ver esta colección con el link.' : 'Solo tú puedes verla.'}
          </small>
        </div>

        {#if error}
          <p class="error-msg">{error}</p>
        {/if}

        <div class="modal-acciones">
          <button type="button" class="btn-secundario" on:click={cerrar}>Cancelar</button>
          <button type="submit" class="btn-primary" disabled={cargando}>
            {cargando ? 'Creando…' : 'Crear'}
          </button>
        </div>
      </form>
    </div>
  </div>
{/if}
