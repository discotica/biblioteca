<script>
  import { createEventDispatcher } from 'svelte';

  export let page = 1;
  export let totalPages = 1;
  export let label = 'Paginación de resultados';

  const dispatch = createEventDispatcher();

  function changePage(next) {
    if (next >= 1 && next <= totalPages && next !== page) dispatch('change', next);
  }
</script>

{#if totalPages > 1}
  <nav class="pagination" aria-label={label}>
    <button class="pagination__nav ui-icon-button" type="button" on:click={() => changePage(page - 1)} disabled={page <= 1} aria-label="Página anterior">‹</button>

    <span class="pagination__status" aria-current="page" aria-label={`Página ${page} de ${totalPages}`}>
      <span aria-hidden="true">Página </span><strong>{page}</strong><span aria-hidden="true"> de {totalPages}</span>
    </span>
    <button class="pagination__nav ui-icon-button" type="button" on:click={() => changePage(page + 1)} disabled={page >= totalPages} aria-label="Página siguiente">›</button>
  </nav>
{/if}
