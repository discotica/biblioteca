<script lang="ts">
  import type { Snippet } from 'svelte';
  import IconButton from './IconButton.svelte';
  import { dialogOverlay } from './overlay';

  type Props = {
    /** Abierto o cerrado. Se cierra con ×, Escape o el fondo (si `dismissible`). */
    open: boolean;
    /** Título visible; es el nombre accesible del diálogo. */
    title: string;
    /** Texto de apoyo bajo el título (aria-describedby). */
    description?: string;
    /** sm 28rem · md 40rem · lg 60rem; nunca más ancho que el viewport menos el margen. */
    size?: 'sm' | 'md' | 'lg';
    /** Tocar el fondo cierra. Escape y × cierran siempre (contrato de overlay.ts). */
    dismissible?: boolean;
    /** Selector del elemento que recibe el foco al abrir; por defecto, el primero enfocable. */
    initialFocus?: string;
    /** A quién devolver el foco al cerrar si no fue el elemento que lo abrió. */
    returnFocus?: HTMLElement | null;
    onclose?: () => void;
    children: Snippet;
    /** Pie fijo bajo el contenido con scroll: acciones del diálogo. */
    footer?: Snippet;
    class?: string;
  };

  let {
    open = $bindable(),
    title,
    description,
    size = 'md',
    dismissible = true,
    initialFocus,
    returnFocus,
    onclose,
    children,
    footer,
    class: className = ''
  }: Props = $props();

  const id = $props.id();
  const titleId = `${id}-title`;
  const descriptionId = `${id}-description`;

  let content = $state<HTMLElement>();
  let body = $state<HTMLElement>();
  let scrollable = $state(false);

  function close() {
    open = false;
    onclose?.();
  }

  // El cuerpo solo entra en el orden de tabulación cuando tiene scroll, para que
  // se pueda desplazar con teclado aunque no contenga nada enfocable.
  $effect(() => {
    if (!body || !content) return;
    const measure = () => (scrollable = body!.scrollHeight > body!.clientHeight + 1);
    const observer = new ResizeObserver(measure);
    observer.observe(body);
    observer.observe(content);
    measure();
    return () => observer.disconnect();
  });
</script>

{#if open}
  <div
    class="ui-dialog"
    role="dialog"
    aria-modal="true"
    aria-labelledby={titleId}
    aria-describedby={description ? descriptionId : undefined}
    use:dialogOverlay={{ onClose: close, initialFocus, returnFocus }}
  >
    <div class="ui-dialog__backdrop" role="presentation" onclick={() => dismissible && close()}></div>
    <div class="ui-dialog__panel ui-dialog__panel--{size} {className}" class:ui-dialog__panel--scrollable={scrollable}>
      <header class="ui-dialog__header">
        <div class="ui-dialog__heading">
          <h2 class="ui-dialog__title" id={titleId}>{title}</h2>
          {#if description}<p class="ui-dialog__description" id={descriptionId}>{description}</p>{/if}
        </div>
        <IconButton icon="cerrar" label="Cerrar" tooltip="bottom" tooltipAlign="end" onclick={close} />
      </header>
      <!-- Región con scroll: enfocable solo cuando desborda, para desplazarla con teclado. -->
      <!-- svelte-ignore a11y_no_noninteractive_tabindex -->
      <div
        class="ui-dialog__body"
        bind:this={body}
        tabindex={scrollable ? 0 : undefined}
        role={scrollable ? 'region' : undefined}
        aria-labelledby={scrollable ? titleId : undefined}
      >
        <div bind:this={content}>{@render children()}</div>
      </div>
      {#if footer}<footer class="ui-dialog__footer">{@render footer()}</footer>{/if}
    </div>
    <!-- Portal de menús y listas abiertos desde el diálogo (layerFor en overlay.ts):
         dentro del diálogo para el foco, fuera del panel para que nada se corte. -->
    <div class="ui-dialog__layer" data-ui-layer></div>
  </div>
{/if}
