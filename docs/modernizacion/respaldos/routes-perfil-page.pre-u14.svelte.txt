<script lang="ts">
  import { onMount } from 'svelte';
  import type { Crate } from '$lib/stores/cratesStore';
  import { supabase } from '$lib/supabase';
  import {
    CONTRIBUTION_STATUS_LABELS,
    getProfileRole,
    loadOwnContributions,
    type Contribution,
    type ProfileRole
  } from '$lib/stores/contributionsStore';
  import PageLayout from '$lib/layouts/PageLayout.svelte';
  import Button from '$lib/ui/Button.svelte';

  export let data: {
    perfil: {
      id: string;
      username: string;
      display_name: string | null;
      bio: string | null;
      created_at: string;
    };
    crates: Crate[];
  };

  $: nombre = data.perfil.display_name || data.perfil.username;
  $: inicial = nombre[0].toUpperCase();
  $: año = new Date(data.perfil.created_at).getFullYear();

  const editorialEnabled = import.meta.env.VITE_EDITORIAL_ENABLED === 'true';
  const adminCatalogEnabled = import.meta.env.VITE_ADMIN_CATALOG_ENABLED === 'true';
  let isOwnProfile = false;
  let role: ProfileRole | '' = '';
  let contributions: Contribution[] = [];
  let contributionsLoading = false;
  let contributionsError = '';

  // Acceso al Estudio (ronda 3, I3): sale del perfil propio de admin y
  // colaborador, no del header. Hasta U15 enlaza a las herramientas actuales.
  $: studioLinks = [
    ...(role === 'admin' && adminCatalogEnabled ? [{ label: 'Catálogo', href: '/admin/catalogo' }] : []),
    ...((role === 'admin' || role === 'colaborador') && editorialEnabled
      ? [{ label: role === 'admin' ? 'Propuestas y reportes' : 'Propuestas', href: '/editorial' }]
      : [])
  ];

  async function loadRole(userId: string) {
    try {
      role = await getProfileRole(userId);
    } catch {
      role = '';
    }
  }

  onMount(async () => {
    const { data: auth } = await supabase.auth.getUser();
    isOwnProfile = auth.user?.id === data.perfil.id;
    if (!isOwnProfile) return;
    void loadRole(data.perfil.id);
    if (!editorialEnabled) return;
    contributionsLoading = true;
    try {
      contributions = await loadOwnContributions(data.perfil.id);
    } catch (cause: any) {
      contributionsError = cause?.message || 'No se pudieron cargar tus propuestas.';
    } finally {
      contributionsLoading = false;
    }
  });

  function date(value: string) {
    return new Intl.DateTimeFormat('es-CO', { dateStyle: 'medium' }).format(new Date(value));
  }
</script>

<svelte:head>
  <title>@{data.perfil.username} — Biblioteca Discótica</title>
</svelte:head>

<PageLayout container="regular" title={nombre}>
  {#snippet actions()}
    {#if isOwnProfile}<Button variant="secondary" href="/cuenta">Editar cuenta</Button>{/if}
  {/snippet}
  {#snippet leading()}
    <div class="avatar" aria-hidden="true">{inicial}</div>
  {/snippet}
  {#snippet meta()}
    <p class="perfil-username">@{data.perfil.username}</p>
    {#if data.perfil.bio}
      <p class="perfil-bio">{data.perfil.bio}</p>
    {/if}
    <p class="perfil-meta">En Discótica desde {año}</p>
  {/snippet}

  {#if studioLinks.length}
    <section class="studio" aria-labelledby="studio-title">
      <div>
        <h2 id="studio-title">Estudio</h2>
        <p>Catálogo, propuestas y reportes.</p>
      </div>
      <div class="studio__links">
        {#each studioLinks as link}
          <Button variant="secondary" href={link.href} iconEnd="flecha-derecha">{link.label}</Button>
        {/each}
      </div>
    </section>
  {/if}

  {#if isOwnProfile && editorialEnabled}
    <section id="mis-aportes" class="contributions-section" aria-labelledby="contributions-title">
      <div class="section-heading">
        <div><p class="eyebrow">Privado</p><h2 id="contributions-title">Mis propuestas</h2></div>
        <a class="ui-button ui-button--secondary new-contribution" href="/sugerir">Nueva propuesta</a>
      </div>
      {#if contributionsLoading}
        <p class="vacio" role="status">Cargando propuestas…</p>
      {:else if contributionsError}
        <p class="contribution-error" role="alert">{contributionsError}</p>
      {:else if contributions.length === 0}
        <div class="empty-contributions"><p>Aún no tienes borradores ni propuestas enviadas.</p><a href="/sugerir">Sugerir un lanzamiento</a></div>
      {:else}
        <ul class="contributions-list">
          {#each contributions as contribution (contribution.id)}
            <li>
              <div class="contribution-main">
                <span class="status status--{contribution.status}">{CONTRIBUTION_STATUS_LABELS[contribution.status]}</span>
                <h3>{contribution.payload?.title || 'Borrador sin título'}</h3>
                <p>{contribution.source_provider ?? 'Fuente'} · Actualizado {date(contribution.updated_at)}</p>
                {#if contribution.reviewer_note}<div class="review-note"><strong>Nota editorial</strong><p>{contribution.reviewer_note}</p></div>{/if}
              </div>
              {#if contribution.status === 'draft' || contribution.status === 'changes_requested'}
                <a class="ui-button ui-button--secondary resume" href={`/sugerir?draft=${contribution.id}`}>{contribution.status === 'draft' ? 'Retomar borrador' : 'Corregir propuesta'}</a>
              {/if}
            </li>
          {/each}
        </ul>
      {/if}
    </section>
  {/if}

  <section id="colecciones" class="crates-seccion">
    <h2 class="seccion-titulo">
      Colecciones públicas
      <span class="seccion-conteo">{data.crates.length}</span>
    </h2>

    {#if data.crates.length === 0}
      <p class="vacio">Este perfil no tiene colecciones públicas todavía.</p>
    {:else}
      <ul class="crates-grid">
        {#each data.crates as crate (crate.id)}
          <li>
            <a class="crate-card" href="/coleccion/{crate.slug}?usuario={data.perfil.username}">
              <h3 class="crate-nombre">{crate.name}</h3>
              {#if crate.description}
                <p class="crate-desc">{crate.description}</p>
              {/if}
              <span class="crate-conteo">
                {crate.item_count ?? 0} {(crate.item_count ?? 0) === 1 ? 'lanzamiento' : 'lanzamientos'}
              </span>
            </a>
          </li>
        {/each}
      </ul>
    {/if}
  </section>
</PageLayout>
