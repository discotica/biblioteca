<script lang="ts">
  // Reportes del Estudio (U15). Es la sección de reportes de la estación
  // editorial de E10, solo para administración, con enlaces al lanzamiento.
  // Su rediseño en bandeja es U17. El layout ya comprobó sesión y rol.
  import { onMount } from 'svelte';
  import { page } from '$app/stores';
  import { Button, LoadingIndicator, Notice } from '$lib/ui';
  import { ToolHeader } from '$lib/layouts';
  import { getStudioContext } from '$lib/components/estudio/context';
  import { studioReleaseHref, studioReportsHref } from '$lib/domain/studio';
  import {
    loadProblemReports,
    updateProblemReport,
    REPORT_CATEGORIES,
    REPORT_STATUSES,
    reportStatusLabel,
    type ProblemReport
  } from '$lib/services/problemReports';

  const studio = getStudioContext();
  const userId = studio.userId;
  const catalogAvailable = studio.sections.includes('catalogo');

  let loaded = false;
  let reports: ProblemReport[] = [];
  let error = '';
  let flash = '';
  let busy = '';
  let reportNotes: Record<string, string> = {};
  let reportStates: Record<string, ProblemReport['status']> = {};

  /** ?lanzamiento=<uuid>: solo los reportes de ese lanzamiento. */
  $: releaseFilter = $page.url.searchParams.get('lanzamiento');
  $: visible = releaseFilter ? reports.filter((report) => report.release_id === releaseFilter) : reports;

  async function load() {
    loaded = false;
    error = '';
    try {
      reports = await loadProblemReports();
    } catch (cause: any) {
      reports = [];
      error = cause?.message || 'No se pudieron cargar los reportes.';
    } finally {
      loaded = true;
    }
  }

  onMount(load);

  async function updateReport(report: ProblemReport) {
    busy = `report:${report.id}`; error = ''; flash = '';
    try {
      const updated = await updateProblemReport(report.id, reportStates[report.id] ?? report.status, userId, reportNotes[report.id] ?? '');
      reports = reports.map((entry) => entry.id === updated.id ? updated : entry);
      flash = 'El reporte fue actualizado.';
      studio.refreshCounts();
    } catch (cause: any) { error = cause?.message || 'No se pudo actualizar el reporte.'; }
    finally { busy = ''; }
  }

  function reportCategory(value: string) { return REPORT_CATEGORIES.find((item) => item[0] === value)?.[1] ?? value; }
</script>

<svelte:head><title>Reportes · Estudio — Discótica</title></svelte:head>

<ToolHeader
  title="Reportes"
  description={releaseFilter
    ? 'Reportes de un lanzamiento.'
    : 'Los adjuntos y el contexto técnico solo los ve administración. Las respuestas se registran internamente; no envían correo.'}
>
  {#snippet actions()}
    {#if releaseFilter}
      <Button size="sm" variant="secondary" href={studioReportsHref()}>Ver todos</Button>
      {#if catalogAvailable}<Button size="sm" variant="secondary" href={studioReleaseHref(releaseFilter)}>Abrir el lanzamiento</Button>{/if}
    {/if}
  {/snippet}
</ToolHeader>

{#if error}<Notice tone="error" class="reportes__notice">{error}</Notice>{/if}
{#if flash}<Notice tone="success" class="reportes__notice">{flash}</Notice>{/if}

{#if !loaded}
  <div class="reportes__loading"><LoadingIndicator label="Cargando reportes" /></div>
{:else}
  <section class="queue" aria-labelledby="reports-heading">
    <div class="queue-heading"><div><p>Soporte privado</p><h2 id="reports-heading">{releaseFilter ? 'Reportes de este lanzamiento' : 'Todos los reportes'}</h2></div><span>{visible.length}</span></div>
    {#if visible.length === 0}<p class="empty">{releaseFilter ? 'Este lanzamiento no tiene reportes.' : 'No hay reportes pendientes.'}</p>{:else}<div class="cards">{#each visible as report (report.id)}
      <article class="card report-card">
        <div class="card-head"><div><span class="status">{reportCategory(report.category)}</span><h3>{reportStatusLabel(report.status)}</h3></div><time>{new Date(report.created_at).toLocaleString('es-CO')}</time></div>
        <p class="report-description">{report.description}</p>
        <dl class="details">
          <div><dt>Ruta</dt><dd>{report.page_url}</dd></div>
          <div><dt>Lanzamiento</dt><dd>
            {#if report.release_id && catalogAvailable}
              <a class="release-link" href={studioReleaseHref(report.release_id)}>Abrir en el Catálogo</a>
              {#if !releaseFilter}· <a class="release-link" href={studioReportsHref(report.release_id)}>Sus reportes</a>{/if}
            {:else}{report.release_id ?? '—'}{/if}
          </dd></div>
          <div><dt>Captura</dt><dd>{report.screenshot_path ? 'Adjunta (privada)' : 'Sin captura'}</dd></div>
          <div><dt>Asignada</dt><dd>{report.assigned_to ? 'Administración' : 'Sin asignar'}</dd></div>
        </dl>
        <label class="note">Estado<select bind:value={reportStates[report.id]}><option value={report.status}>{reportStatusLabel(report.status)}</option>{#each REPORT_STATUSES as [state, label]}{#if state !== report.status}<option value={state}>{label}</option>{/if}{/each}</select></label>
        <label class="note">Respuesta interna<textarea bind:value={reportNotes[report.id]} rows="3" placeholder="Clasificación o respuesta para el equipo"></textarea></label>
        <button class="ui-button ui-button--primary" on:click={() => updateReport(report)} disabled={busy === `report:${report.id}`}>{busy === `report:${report.id}` ? 'Guardando…' : 'Guardar clasificación'}</button>
      </article>
    {/each}</div>{/if}
  </section>
{/if}
