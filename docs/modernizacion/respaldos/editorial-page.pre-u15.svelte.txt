<script lang="ts">
  import PageLayout from '$lib/layouts/PageLayout.svelte';
  import { onMount, tick } from 'svelte';
  import { dialogOverlay } from '$lib/ui/overlay';
  import Icon from '$lib/ui/Icon.svelte';
  import { supabase } from '$lib/supabase';
  import {
    CONTRIBUTION_STATUS_LABELS,
    RELEASE_TYPE_OPTIONS,
    getProfileRole,
    loadEditorialQueue,
    prepareContributionForAdmin,
    reviewContribution,
    type Contribution,
    type ProfileRole
  } from '$lib/stores/contributionsStore';
  import { loadProblemReports, updateProblemReport, REPORT_CATEGORIES, REPORT_STATUSES, reportStatusLabel, type ProblemReport } from '$lib/services/problemReports';

  const editorialEnabled = import.meta.env.VITE_EDITORIAL_ENABLED === 'true';
  let role: ProfileRole | '' = '';
  let userId = '';
  let loaded = false;
  let queue: Contribution[] = [];
  let error = '';
  let flash = '';
  let busy = '';
  let notes: Record<string, string> = {};
  let approvalTarget: Contribution | null = null;
  let reports: ProblemReport[] = [];
  let reportNotes: Record<string, string> = {};
  let reportStates: Record<string, ProblemReport['status']> = {};

  $: communityQueue = queue.filter((item) => item.queue_kind === 'community' && item.status === 'pending');
  $: priorityQueue = queue.filter((item) => item.queue_kind === 'collaborator' || item.status === 'ready_for_admin');

  onMount(async () => {
    if (!editorialEnabled) { loaded = true; return; }
    try {
      const { data, error: authError } = await supabase.auth.getUser();
      if (authError) throw authError;
      userId = data.user?.id ?? '';
      if (!userId) return;
      role = await getProfileRole(userId);
      if (role === 'admin' || role === 'colaborador') queue = await loadEditorialQueue();
      if (role === 'admin') reports = await loadProblemReports();
    } catch (cause: any) {
      error = cause?.message || 'No se pudo cargar la estación editorial.';
    } finally {
      loaded = true;
    }
  });

  function releaseType(value: string) {
    return RELEASE_TYPE_OPTIONS.find((option) => option.value === value)?.label ?? (value || 'Sin confirmar');
  }

  async function updateReport(report: ProblemReport) {
    busy = `report:${report.id}`; error = ''; flash = '';
    try {
      const updated = await updateProblemReport(report.id, reportStates[report.id] ?? report.status, userId, reportNotes[report.id] ?? '');
      reports = reports.map((entry) => entry.id === updated.id ? updated : entry);
      flash = 'El reporte fue actualizado.';
    } catch (cause: any) { error = cause?.message || 'No se pudo actualizar el reporte.'; }
    finally { busy = ''; }
  }

  function reportCategory(value: string) { return REPORT_CATEGORIES.find((item) => item[0] === value)?.[1] ?? value; }

  function display(value: unknown) {
    if (Array.isArray(value)) return value.length ? value.join(', ') : '—';
    return typeof value === 'string' && value.trim() ? value : '—';
  }

  function detected(item: Contribution, key: 'title' | 'artists' | 'type' | 'release_date') {
    return display(item.detected_metadata?.[key]);
  }

  function confirmed(item: Contribution, key: 'title' | 'artists' | 'type' | 'release_date') {
    if (key === 'artists') return display(item.payload.primary_artists);
    if (key === 'type') return releaseType(item.payload.release_type);
    return display(item.payload[key]);
  }

  function differs(item: Contribution, key: 'title' | 'artists' | 'type' | 'release_date') {
    const before = detected(item, key).toLocaleLowerCase('es');
    const after = confirmed(item, key).toLocaleLowerCase('es');
    return before !== '—' && before !== after;
  }

  async function preparar(item: Contribution) {
    busy = item.id;
    error = '';
    flash = '';
    try {
      const updated = await prepareContributionForAdmin(item.id, item.payload, notes[item.id] ?? '');
      queue = role === 'colaborador'
        ? queue.filter((entry) => entry.id !== item.id)
        : queue.map((entry) => entry.id === item.id ? updated : entry);
      flash = `${item.payload.title || 'La propuesta'} quedó lista para administración.`;
    } catch (cause: any) {
      error = cause?.message || 'No se pudo preparar la propuesta.';
    } finally {
      busy = '';
    }
  }

  async function decide(item: Contribution, action: 'approved' | 'rejected' | 'changes_requested') {
    error = '';
    flash = '';
    const note = notes[item.id]?.trim() ?? '';
    if (action !== 'approved' && !note) {
      error = action === 'rejected' ? 'Escribe una nota antes de rechazar.' : 'Escribe una nota para solicitar cambios.';
      await tick();
      document.getElementById(`note-${item.id}`)?.focus();
      return;
    }
    busy = item.id;
    try {
      const updated = await reviewContribution(item.id, action, note);
      queue = queue.filter((entry) => entry.id !== item.id);
      flash = `${item.payload.title || 'La propuesta'}: ${CONTRIBUTION_STATUS_LABELS[updated.status]}.`;
      approvalTarget = null;
    } catch (cause: any) {
      error = cause?.message || 'No se pudo registrar la decisión.';
      if (action === 'approved') approvalTarget = null;
    } finally {
      busy = '';
    }
  }
</script>

<svelte:head><title>Estación editorial — Discótica</title></svelte:head>

<PageLayout eyebrow="Operación editorial" title="Estación editorial" description="Las colas comunitaria y prioritaria se revisan por separado. La base de datos conserva la autorización y cada transición.">

  {#if !editorialEnabled}
    <section class="gate"><h2>Estación deshabilitada</h2><p>La función editorial todavía no está habilitada en este entorno.</p></section>
  {:else if !loaded}
    <p class="loading" role="status">Cargando permisos y colas…</p>
  {:else if !userId}
    <section class="gate"><h2>Inicia sesión</h2><p>Necesitas una sesión editorial para consultar esta ruta.</p><a class="ui-button ui-button--secondary link-button" href="/auth/login?redirect=/editorial">Ir a iniciar sesión</a></section>
  {:else if role !== 'admin' && role !== 'colaborador'}
    <section class="gate" role="alert"><h2>Sin acceso editorial</h2><p>Tu cuenta no puede consultar ni actuar sobre estas colas.</p></section>
  {:else}
    {#if error}<p class="alert" role="alert">{error}</p>{/if}
    {#if flash}<p class="flash" role="status">{flash}</p>{/if}

    {#if role === 'admin'}
      <section class="queue" aria-labelledby="priority-heading">
        <div class="queue-heading"><div><p>Revisión final</p><h2 id="priority-heading">Cola prioritaria</h2></div><span>{priorityQueue.length}</span></div>
        <p class="queue-help">Aportes de colaboradores y propuestas comunitarias ya preparadas.</p>
        {#if priorityQueue.length === 0}<p class="empty">No hay propuestas en la cola prioritaria.</p>{:else}<div class="cards">{#each priorityQueue as item (item.id)}{@render ReviewCard(item)}{/each}</div>{/if}
      </section>

      <section class="queue" aria-labelledby="reports-heading">
        <div class="queue-heading"><div><p>Soporte privado</p><h2 id="reports-heading">Reportes</h2></div><span>{reports.length}</span></div>
        <p class="queue-help">Los adjuntos y el contexto técnico sólo son visibles para administración. Las respuestas se registran internamente; no envían correo.</p>
        {#if reports.length === 0}<p class="empty">No hay reportes pendientes.</p>{:else}<div class="cards">{#each reports as report (report.id)}
          <article class="card report-card">
            <div class="card-head"><div><span class="status">{reportCategory(report.category)}</span><h3>{reportStatusLabel(report.status)}</h3></div><time>{new Date(report.created_at).toLocaleString('es-CO')}</time></div>
            <p class="report-description">{report.description}</p>
            <dl class="details"><div><dt>Ruta</dt><dd>{report.page_url}</dd></div><div><dt>Lanzamiento</dt><dd>{report.release_id ?? '—'}</dd></div><div><dt>Captura</dt><dd>{report.screenshot_path ? 'Adjunta (privada)' : 'Sin captura'}</dd></div><div><dt>Asignada</dt><dd>{report.assigned_to ? 'Administración' : 'Sin asignar'}</dd></div></dl>
            <label class="note">Estado<select bind:value={reportStates[report.id]}><option value={report.status}>{reportStatusLabel(report.status)}</option>{#each REPORT_STATUSES as [state, label]}{#if state !== report.status}<option value={state}>{label}</option>{/if}{/each}</select></label>
            <label class="note">Respuesta interna<textarea bind:value={reportNotes[report.id]} rows="3" placeholder="Clasificación o respuesta para el equipo"></textarea></label>
            <button class="ui-button ui-button--primary" on:click={() => updateReport(report)} disabled={busy === `report:${report.id}`}>{busy === `report:${report.id}` ? 'Guardando…' : 'Guardar clasificación'}</button>
          </article>
        {/each}</div>{/if}
      </section>
    {:else if priorityQueue.length}
      <section class="queue" aria-labelledby="own-priority-heading">
        <div class="queue-heading"><div><p>Sin auto-revisión</p><h2 id="own-priority-heading">Mis aportes prioritarios</h2></div><span>{priorityQueue.length}</span></div>
        <p class="queue-help">Tus propuestas pasan directamente a administración. Aquí no aparecen controles de revisión.</p>
        <div class="cards">{#each priorityQueue as item (item.id)}{@render ReviewCard(item)}{/each}</div>
      </section>
    {/if}

    <section class="queue" aria-labelledby="community-heading">
      <div class="queue-heading"><div><p>{role === 'admin' ? 'Decisión administrativa' : 'Preparación colaborativa'}</p><h2 id="community-heading">Cola comunitaria</h2></div><span>{communityQueue.length}</span></div>
      <p class="queue-help">{role === 'admin' ? 'Propuestas pendientes creadas por la comunidad.' : 'Puedes preparar aportes comunitarios ajenos para la revisión final de administración.'}</p>
      {#if communityQueue.length === 0}<p class="empty">No hay propuestas comunitarias pendientes.</p>{:else}<div class="cards">{#each communityQueue as item (item.id)}{@render ReviewCard(item)}{/each}</div>{/if}
    </section>
  {/if}
</PageLayout>

{#snippet ReviewCard(item: Contribution)}
  <article class="card">
    <div class="card-head">
      <div><span class="status">{CONTRIBUTION_STATUS_LABELS[item.status]}</span><h3>{item.payload.title || 'Propuesta sin título'}</h3></div>
      <span class="provider">{item.source_provider ?? 'Fuente'}</span>
    </div>
    <a class="source-link" href={item.source_url} target="_blank" rel="noreferrer">Abrir fuente original<span class="source-link__icon"><Icon name="flecha-externa" size="em" /></span></a>

    <div class="comparison" aria-label="Comparación entre detección y datos confirmados">
      <div class="comparison-head"><span>Campo</span><span>Detectado</span><span>Confirmado</span></div>
      {#each [['title','Título'], ['artists','Principales'], ['type','Tipo'], ['release_date','Fecha']] as row}
        <div class:changed={differs(item, row[0] as any)} class="comparison-row"><strong>{row[1]}</strong><span>{detected(item, row[0] as any)}</span><span>{confirmed(item, row[0] as any)}</span></div>
      {/each}
    </div>

    <dl class="details">
      <div><dt>Colaboradores</dt><dd>{display(item.payload.collaborators)}</dd></div>
      <div><dt>Géneros</dt><dd>{display(item.payload.genres)}</dd></div>
      <div><dt>Países</dt><dd>{display(item.payload.countries)}</dd></div>
      <div><dt>Sello</dt><dd>{display(item.payload.label)}</dd></div>
    </dl>
    {#if item.detected_metadata?.cover_url}<div class="cover"><img src={String(item.detected_metadata.cover_url)} alt="Portada detectada para {item.payload.title}" /><p>Evidencia detectada; verificar antes de publicar.</p></div>{/if}
    {#if item.payload.team_note}<div class="author-note"><strong>Nota para el equipo</strong><p>{item.payload.team_note}</p></div>{/if}

    {#if role === 'colaborador' && item.queue_kind === 'community' && item.status === 'pending' && item.author_id !== userId}
      <label class="note" for={`note-${item.id}`}>Nota interna <span>Opcional</span><textarea id={`note-${item.id}`} bind:value={notes[item.id]} rows="3" placeholder="Evidencia o contexto para administración"></textarea></label>
      <button class="ui-button ui-button--primary" on:click={() => preparar(item)} disabled={busy === item.id}>{busy === item.id ? 'Preparando…' : 'Marcar lista para administración'}</button>
    {:else if role === 'admin'}
      <label class="note" for={`note-${item.id}`}>Nota para la persona autora <span>Obligatoria al pedir cambios o rechazar</span><textarea id={`note-${item.id}`} bind:value={notes[item.id]} rows="3"></textarea></label>
      <div class="actions">
        <button class="ui-button ui-button--primary" on:click={() => approvalTarget = item} disabled={busy === item.id}>Revisar publicación</button>
        <button class="ui-button ui-button--secondary" on:click={() => decide(item, 'changes_requested')} disabled={busy === item.id}>Solicitar cambios</button>
        <button class="ui-button ui-button--destructive" on:click={() => decide(item, 'rejected')} disabled={busy === item.id}>Rechazar</button>
      </div>
    {:else}
      <p class="no-actions">Pendiente de revisión administrativa.</p>
    {/if}
  </article>
{/snippet}

{#if approvalTarget}
  <div class="backdrop" role="presentation" on:click|self={() => approvalTarget = null}>
    <div class="approval-dialog" role="dialog" aria-modal="true" aria-labelledby="approval-title" aria-describedby="approval-description" use:dialogOverlay={{ onClose: () => approvalTarget = null }}>
      <p class="kicker">Publicación final</p><h2 id="approval-title">{approvalTarget.payload.title}</h2>
      <p id="approval-description">Confirma el release que se publicará de forma transaccional.</p>
      <dl class="summary">
        <div><dt>Principales</dt><dd>{display(approvalTarget.payload.primary_artists)}</dd></div>
        <div><dt>Colaboradores</dt><dd>{display(approvalTarget.payload.collaborators)}</dd></div>
        <div><dt>Tipo</dt><dd>{releaseType(approvalTarget.payload.release_type)}</dd></div>
        <div><dt>Fecha</dt><dd>{approvalTarget.payload.release_date}</dd></div>
        <div><dt>Géneros</dt><dd>{display(approvalTarget.payload.genres)}</dd></div>
        <div><dt>Países</dt><dd>{display(approvalTarget.payload.countries)}</dd></div>
      </dl>
      <div class="dialog-actions"><button class="ui-button ui-button--secondary" data-overlay-initial-focus on:click={() => approvalTarget = null}>Volver</button><button class="ui-button ui-button--primary" on:click={() => approvalTarget && decide(approvalTarget, 'approved')} disabled={busy === approvalTarget.id}>{busy === approvalTarget.id ? 'Publicando…' : 'Aprobar y publicar'}</button></div>
    </div>
  </div>
{/if}
