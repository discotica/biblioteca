<script lang="ts">
  import type { Snippet } from 'svelte';
  import Icon from './Icon.svelte';

  type Props = {
    checked?: boolean;
    /** Texto de la casilla. Puede llevar un enlace. */
    children: Snippet;
    name?: string;
    id?: string;
    /** Mensaje de error propio (no la burbuja nativa del navegador). */
    error?: string;
    disabled?: boolean;
    class?: string;
    element?: HTMLInputElement;
    [attribute: string]: unknown;
  };

  let {
    checked = $bindable(false),
    children,
    name,
    id,
    error,
    disabled = false,
    class: className = '',
    element = $bindable(),
    ...rest
  }: Props = $props();

  const uid = $props.id();
  const inputId = $derived(id ?? `${uid}-checkbox`);
  const errorId = `${uid}-error`;
</script>

<div class="ui-checkbox {className}" class:ui-checkbox--error={!!error} class:ui-checkbox--disabled={disabled}>
  <label class="ui-checkbox__label" for={inputId}>
    <span class="ui-checkbox__control">
      <input
        bind:this={element}
        bind:checked
        class="ui-checkbox__input"
        type="checkbox"
        id={inputId}
        {name}
        {disabled}
        aria-invalid={error ? 'true' : undefined}
        aria-describedby={error ? errorId : undefined}
        {...rest}
      />
      <span class="ui-checkbox__box" aria-hidden="true">{#if checked}<Icon name="check" size="sm" />{/if}</span>
    </span>
    <span class="ui-checkbox__text">{@render children()}</span>
  </label>
  {#if error}<p class="ui-checkbox__error" id={errorId}>{error}</p>{/if}
</div>

<style>

  /* La caja mide un icono; el input invisible cubre toda la fila (44 px de alto
     como mínimo), así que el área táctil es la fila completa. */
  .ui-checkbox__control {
    display: inline-flex;
    flex-shrink: 0;
    margin-top: calc((1.5em - var(--icon-md)) / 2);
  }
  .ui-checkbox__input {
    position: absolute;
    inset: 0;
    width: 100%;
    height: 100%;
    margin: 0;
    opacity: 0;
    cursor: inherit;
  }
  .ui-checkbox__box {
    display: inline-flex;

  }
  .ui-checkbox__input:checked + .ui-checkbox__box {

  }
  .ui-checkbox__label:hover .ui-checkbox__input:not(:disabled) + .ui-checkbox__box { border-color: var(--color-acento-border); }

  }
