<script lang="ts">
  type Props = {
    /** Texto que se anuncia y, si `showLabel`, se ve en mayúsculas junto a los puntos. */
    label?: string;
    showLabel?: boolean;
    size?: 'sm' | 'md';

    decorative?: boolean;

  let { label = 'Cargando', showLabel = true, size = 'md', decorative = false, class: className = '' }: Props = $props();
</script>

     horizontal, como pidió la ronda 3. Se combina con Skeleton. -->

  class="ui-loading ui-loading--{size} {className}"
  role={decorative ? undefined : 'status'}
  aria-hidden={decorative ? 'true' : undefined}
>

    <span class="ui-loading__dot"></span>
    <span class="ui-loading__dot"></span>
    <span class="ui-loading__dot"></span>
  </span>
  {#if !decorative}
    <span class="ui-loading__label" class:sr-only={!showLabel}>{label}</span>
  {/if}
</span>
