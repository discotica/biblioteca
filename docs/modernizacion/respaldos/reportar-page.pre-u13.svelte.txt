<script lang="ts">
  import PageLayout from '$lib/layouts/PageLayout.svelte';
  import { goto } from '$app/navigation';
  import { page } from '$app/stores';
  import { onMount } from 'svelte';
  import { user } from '$lib/stores/authStore';
  import { createProblemReport, REPORT_CATEGORIES, type ReportCategory, validateScreenshot } from '$lib/services/problemReports';

  let category: ReportCategory = 'technical_bug';
  let description = '';
  let screenshot: File | null = null;
  let error = '';
  let notice = '';
  let busy = false;
  let releaseId: string | null = null;
  $: returnTo = `${$page.url.pathname}${$page.url.search}`;

  onMount(() => {
    const candidate = $page.url.searchParams.get('release_id');
    releaseId = candidate && /^[0-9a-f]{8}-[0-9a-f-]{27}$/i.test(candidate) ? candidate : null;
    if (!$user) void goto(`/auth/login?redirect=${encodeURIComponent(returnTo)}`);
  });

  function chooseFile(event: Event) {
    error = '';
    const next = (event.currentTarget as HTMLInputElement).files?.[0] ?? null;
    try { validateScreenshot(next); screenshot = next; } catch (cause: any) { screenshot = null; error = cause?.message ?? 'No se pudo usar esta captura.'; }
  }

  async function submit() {
    error = ''; notice = '';
    if (!description.trim()) { error = 'Describe el problema para poder investigarlo.'; return; }
    busy = true;
    try {
      const result = await createProblemReport({ category, description, releaseId, screenshot });
      notice = result.screenshotError ?? 'Tu reporte fue enviado. Gracias por ayudarnos a mejorar la biblioteca.';
      description = ''; screenshot = null;
      const input = document.getElementById('report-screenshot') as HTMLInputElement | null;
      if (input) input.value = '';
    } catch (cause: any) { error = cause?.message ?? 'No se pudo enviar el reporte. Intenta de nuevo.'; }
    finally { busy = false; }
  }
</script>

<svelte:head><title>Reportar un problema — Discótica</title></svelte:head>

<PageLayout container="narrow" eyebrow="Soporte" title="Reportar un problema" description="Cuéntanos sobre un error de catálogo, un problema técnico o una mejora. Este canal requiere una sesión para proteger tu reporte.">
  {#if releaseId}<p class="context">Este reporte incluye el lanzamiento desde el que llegaste.</p>{/if}
  <form on:submit|preventDefault={submit}>
    <label>Categoría<select bind:value={category}>{#each REPORT_CATEGORIES as option}<option value={option[0]}>{option[1]}</option>{/each}</select></label>
    <label>Descripción<textarea bind:value={description} required maxlength="5000" rows="7" placeholder="¿Qué ocurrió y qué esperabas que pasara?"></textarea></label>
    <label>Captura <span class="optional">opcional, JPEG/PNG/WebP hasta 5 MB</span><input id="report-screenshot" type="file" accept="image/jpeg,image/png,image/webp" on:change={chooseFile} /></label>
    <aside><strong>Contexto técnico incluido</strong><span>ruta, lanzamiento si aplica, fecha, versión, fuente de catálogo, navegador, viewport, idioma y conexión. Nunca incluimos cookies, tokens, contraseñas ni el contenido de otros formularios.</span></aside>
    {#if error}<p class="error" role="alert">{error}</p>{/if}{#if notice}<p class="notice" role="status">{notice}</p>{/if}
    <button class="ui-button ui-button--primary" disabled={busy}>{busy ? 'Enviando…' : 'Enviar reporte'}</button>
  </form>
</PageLayout>
