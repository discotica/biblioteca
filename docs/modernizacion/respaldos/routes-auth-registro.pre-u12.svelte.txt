<script lang="ts">
  import AccountLayout from '$lib/layouts/AccountLayout.svelte';
  import { goto } from '$app/navigation';
  import { page } from '$app/stores';
  import { supabase } from '$lib/supabase';
  let email = '';
  let password = '';
  let confirmPassword = '';
  let username = '';
  let displayName = '';
  let newsletterOptIn = false;
  let privacidadAceptada = false;
  let error = '';
  let mensajeExito = '';
  let cargando = false;
  let usernameDisponible: boolean | null = null;
  let checkingUsername = false;
  const usernameRegex = /^[a-z0-9_-]{3,30}$/;
  const usernamePattern = '[a-z0-9_\\-]{3,30}';
  async function verificarUsername() {
    usernameDisponible = null;
    if (!usernameRegex.test(username)) return;
    checkingUsername = true;
    const { data } = await supabase
      .from('profiles')
      .select('id')
      .eq('username', username)
      .maybeSingle();
    checkingUsername = false;
    usernameDisponible = data === null;
  }
  async function registrar() {
    error = '';
    if (password !== confirmPassword) {
      error = 'Las contraseñas no coinciden.';
      return;
    }
    if (!usernameRegex.test(username)) {
      error = 'El nombre de usuario solo puede tener letras minúsculas, números, guiones y guiones bajos (3–30 caracteres).';
      return;
    }
    if (usernameDisponible === false) {
      error = 'Este nombre de usuario ya está en uso.';
      return;
    }
    if (!privacidadAceptada) {
      error = 'Debes aceptar la política de privacidad para continuar.';
      return;
    }
    cargando = true;
    const { error: err } = await supabase.auth.signUp({
      email,
      password,
      options: {
        data: {
          username,
          display_name: displayName || username,
          newsletter_opt_in: newsletterOptIn
        },
        emailRedirectTo: `${window.location.origin}/auth/reset?flow=signup`
      }
    });
    cargando = false;
    if (err) {
      if (err.message.includes('already registered') || err.message.includes('already been registered')) {
        error = 'Este email ya tiene una cuenta registrada.';
      } else {
        error = `Error: ${err.message}`;
      }
      return;
    }
    mensajeExito = 'Revisa tu correo para confirmar tu cuenta.';
  }
</script>
<svelte:head>
  <title>Crear cuenta — Biblioteca Discótica</title>
</svelte:head>
<AccountLayout title="Crear cuenta">
    {#if mensajeExito}
      <div class="exito-msg">
        <strong>{mensajeExito}</strong>
        <p>Una vez que confirmes tu email puedes entrar y crear tus colecciones.</p>
        <a href={`/auth/login?redirect=${encodeURIComponent($page.url.searchParams.get('redirect') ?? '')}`}>Ir a iniciar sesión</a>
      </div>
    {:else}
      <form on:submit|preventDefault={registrar} class="auth-form">
        <label class="campo">
          <span>Email</span>
          <input type="email" bind:value={email} autocomplete="email" required placeholder="tu@correo.com" />
        </label>
        <label class="campo">
          <span>Nombre de usuario</span>
          <div class="username-wrap">
            <input
              type="text"
              bind:value={username}
              on:blur={verificarUsername}
              autocomplete="username"
              required
              placeholder="ej: cumbiero_total"
              pattern={usernamePattern}
              title="Solo minúsculas, números, - y _ (3–30 caracteres)"
            />
            {#if checkingUsername}
              <span class="username-hint">Verificando…</span>
            {:else if usernameDisponible === true}
              <span class="username-hint disponible">Disponible</span>
            {:else if usernameDisponible === false}
              <span class="username-hint ocupado">Ya está en uso</span>
            {/if}
          </div>
          <small class="campo-ayuda">Solo minúsculas, números, - y _ · 3–30 caracteres · No podrás cambiarlo después</small>
        </label>
        <label class="campo">
          <span>Nombre para mostrar <em class="opcional">(opcional)</em></span>
          <input type="text" bind:value={displayName} autocomplete="name" placeholder="Como quieres que te vean" maxlength="60" />
        </label>
        <label class="campo">
          <span>Contraseña</span>
          <input type="password" bind:value={password} autocomplete="new-password" required minlength="8" />
        </label>
        <label class="campo">
          <span>Confirmar contraseña</span>
          <input type="password" bind:value={confirmPassword} autocomplete="new-password" required minlength="8" />
        </label>
        <label class="campo-check">
          <input type="checkbox" bind:checked={newsletterOptIn} />
          <span>Quiero recibir novedades y ofertas de Discótica por email.</span>
        </label>
        <label class="campo-check campo-check--requerido">
          <input type="checkbox" bind:checked={privacidadAceptada} required />
          <span>
            Leí y acepto la <a href="/privacidad" target="_blank">política de privacidad</a>.
          </span>
        </label>
        {#if error}
          <p class="error-msg">{error}</p>
        {/if}
        <button type="submit" class="btn-primary" disabled={cargando}>
          {cargando ? 'Creando cuenta…' : 'Crear cuenta'}
        </button>
      </form>
    {/if}
    {#snippet footer()}
      {#if !mensajeExito}
        <span>¿Ya tienes cuenta?</span>
        <a href="/auth/login">Entrar</a>
      {/if}
    {/snippet}
</AccountLayout>
